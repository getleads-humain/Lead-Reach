   116→function createInitialPipelineState(): PipelineState {
   117→  // Create one entry per 8-agent display name
   118→  // Each agent's persona field is set via AGENT_8_MAP
   119→  const idleState = (persona: AgentPersona): AgentState => ({
   120→    persona,
   121→    status: 'idle',
   122→    currentStep: '',
   123→    progress: 0,
   124→    startedAt: null,
   125→    completedAt: null,
   126→    thinkTimeMs: null,
   127→  });
   128→
   129→  const agents: Record<string, AgentState> = {
   130→    atlas:  idleState(AGENT_8_MAP['atlas']),
   131→    scout:  idleState(AGENT_8_MAP['scout']),
   132→    forge:  idleState(AGENT_8_MAP['forge']),
   133→    sage:   idleState(AGENT_8_MAP['sage']),
   134→    judge:  idleState(AGENT_8_MAP['judge']),
   135→    bard:   idleState(AGENT_8_MAP['bard']),
   136→    flow:   idleState(AGENT_8_MAP['flow']),
   137→    echo:   idleState(AGENT_8_MAP['echo']),
   138→  };
   139→  return {
   140→    phase: 'idle',
   141→    thinkStartTime: null,
   142→    totalThinkTimeMs: null,
   143→    agents,
   144→    commLog: [],
   145→    currentStep: '',
   146→    overallProgress: 0,
   147→  };
   148→}

   153→
   154→export async function processWithOrchestrator(
   155→  userMessage: string,
   156→  context?: ConversationContext,
   157→  forceIntent?: UserIntent,
   158→  onEvent?: OrchestratorCallback,
   159→  resumeFrom?: PipelineCheckpoint,
   160→): Promise<{
   161→  message: AgentMessage;
   162→  updatedContext: ConversationContext;
   163→  suggestedActions: SuggestedAction[];
   164→  pipelineState: PipelineState;
   165→}> {
   166→  const pipelineState = createInitialPipelineState();
   167→  const startTime = Date.now();
   168→
   169→  try {
   170→    return await processWithOrchestratorInner(userMessage, context, forceIntent, pipelineState, startTime, onEvent, resumeFrom);
   171→  } catch (error) {
   172→    console.error('[Orchestrator] FATAL:', error);
   173→
   174→    const errorMsg = error instanceof Error ? error.message : 'Unknown error';
   175→    pipelineState.phase = 'error';
   176→
   177→    return {
   178→      message: {
   179→        id: `agent-fallback-${Date.now()}`,
   180→        role: 'assistant',
   181→        content: `I encountered an error while processing your request. Let me try a different approach.\n\n**Error:** ${errorMsg.slice(0, 150)}\n\nYou can try:\n• Rephrasing your query (e.g., "Research Stripe")\n• Asking a more specific question`,
   182→        timestamp: new Date(),
   183→        persona: 'navigator',
   184→        thinking: {
   185→          persona: 'navigator',
   186→          intent: 'converse',
   187→          reasoning: `Fallback: ${errorMsg.slice(0, 100)}`,
   188→          plan: ['Error recovery'],
   189→          confidence: 0.1,
   190→        },
   191→        actions: [{ type: 'converse', label: 'Error', status: 'failed', message: errorMsg.slice(0, 100) }],
   192→      },
   193→      updatedContext: context || { recentProspects: [], activeICP: null, lastIntent: null, lastPersona: null, userPreferences: {} },
   194→      suggestedActions: [
   195→        { label: 'Try Again', prompt: userMessage, icon: 'RefreshCw' },
   196→        { label: 'Help', prompt: 'What can you do?', icon: 'Lightbulb' },
   197→      ],
   198→      pipelineState,
   199→    };
   200→  }
   201→}

   211→// ============================================================
   212→// Helper: send agent communication message
   213→// Uses 8-agent display names (atlas, scout, forge, etc.) for
   214→// correct UI rendering in the comm log.
   215→// ============================================================
   216→
   217→function sendCommMsg(
   218→  pipelineState: PipelineState,
   219→  from: string,
   220→  to: string,
   221→  type: AgentCommMessage['type'],
   222→  content: string,
   223→  data?: Record<string, unknown>,
   224→  onEvent?: OrchestratorCallback,
   225→): void {
   226→  const msg: AgentCommMessage = {
   227→    id: `comm-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`,
   228→    from, to, type, content, data, timestamp: Date.now(),
   229→  };
   230→  pipelineState.commLog.push(msg);
   231→  emit(onEvent, { type: 'agent_comm', data: msg });
   232→}
   233→
   234→// ============================================================
   235→// Helper: update agent state
   236→// Uses the 8-agent display name as the key in pipelineState.agents,
   237→// and sets the persona field from AGENT_8_MAP for backward compat.
   238→// ============================================================
   239→
   240→function updateAgentState(
   241→  pipelineState: PipelineState,
   242→  agentKey: string,  // 8-agent display name (atlas, scout, forge, etc.)
   243→  update: Partial<AgentState>,
   244→  onEvent?: OrchestratorCallback,
   245→): void {
   246→  // Ensure persona is set from the mapping if not already provided
   247→  if (!update.persona) {
   248→    update.persona = AGENT_8_MAP[agentKey] || 'navigator';
   249→  }
   250→  const existing = pipelineState.agents[agentKey];
   251→  if (existing) {
   252→    pipelineState.agents[agentKey] = { ...existing, ...update };
   253→  } else {
   254→    // Create the entry if it doesn't exist (safety net)
   255→    pipelineState.agents[agentKey] = {
   256→      persona: update.persona,
   257→      status: 'idle',
   258→      currentStep: '',
   259→      progress: 0,
   260→      startedAt: null,
   261→      completedAt: null,
   262→      thinkTimeMs: null,
   263→      ...update,
   264→    };
   265→  }
   266→  emit(onEvent, { type: 'agent_status', data: { agent: agentKey, state: pipelineState.agents[agentKey] } });
   267→}

   269→// ============================================================
   270→// Helper: build rich fallback response from structured data
   271→// without needing an LLM call. Covers all intent types.
   272→// ============================================================
   273→
   274→function buildRichFallbackResponse(params: {
   275→  intent: UserIntent;
   276→  prospect?: ProspectResult;
   277→  icp?: ICPResult;
   278→  market?: MarketResult;
   279→  score?: ScoreResult;
   280→  outreach?: OutreachResult;
   281→  userMessage: string;
   282→}): string {
   283→  const { intent, prospect, icp, market, score, outreach, userMessage } = params;
   284→
   285→  // Prospect-based responses
   286→  if (prospect) {
   287→    return buildFallbackResponse(prospect, intent);
   288→  }
   289→
   290→  // Market analysis fallback
   291→  if (market) {
   292→    const parts: string[] = ['## Market Analysis'];
   293→    if (market.summary) parts.push(market.summary);
   294→    if (market.keyFindings?.length) {
   295→      parts.push('\n**Key Findings:**');
   296→      market.keyFindings.slice(0, 5).forEach(f => parts.push(`- ${f}`));
   297→    }
   298→    if (market.competitors?.length) {
   299→      parts.push('\n**Competitors:**');
   300→      market.competitors.slice(0, 5).forEach(c => parts.push(`- **${c.name}**: ${c.description?.slice(0, 80) || 'No description'}`));
   301→    }
   302→    if (market.trends?.length) {
   303→      parts.push('\n**Trends:**');
   304→      market.trends.slice(0, 3).forEach(t => parts.push(`- ${t}`));
   305→    }
   306→    return parts.join('\n\n');
   307→  }
   308→
   309→  // Score fallback
   310→  if (score) {
   311→    const parts: string[] = ['## Lead Score'];
   312→    parts.push(`**Overall Score:** ${score.overallScore}/100 — **${score.tier.toUpperCase()}** tier`);
   313→    parts.push(`**Recommendation:** ${score.recommendation}`);
   314→    if (score.dimensions) {
   315→      const dims = Object.entries(score.dimensions) as [string, { score: number; reasoning: string }][];
   316→      parts.push('\n**Dimension Breakdown:**');
   317→      dims.forEach(([key, val]) => {
   318→        parts.push(`- **${key}**: ${val.score}/100 — ${val.reasoning?.slice(0, 60) || ''}`);
   319→      });
   320→    }
   321→    return parts.join('\n\n');
   322→  }
   323→
   324→  // Outreach fallback
   325→  if (outreach) {
   326→    const parts: string[] = ['## Outreach Message'];
   327→    if (outreach.subject) parts.push(`**Subject:** ${outreach.subject}`);
   328→    parts.push(`**Channel:** ${outreach.channel}`);
   329→    parts.push(`**Tone:** ${outreach.tone}`);
   330→    parts.push(`\n${outreach.body}`);
   331→    if (outreach.personalizationHooks?.length) {
   332→      parts.push('\n**Personalization Hooks:**');
   333→      outreach.personalizationHooks.forEach(h => parts.push(`- ${h}`));
   334→    }
   335→    return parts.join('\n\n');
   336→  }
   337→
   338→  // ICP fallback
   339→  if (icp) {
   340→    const parts: string[] = ['## Ideal Customer Profile'];
   341→    parts.push(`**${icp.name}**`);
   342→    if (icp.description) parts.push(icp.description);
   343→    if (icp.firmographic) {
   344→      parts.push(`\n**Firmographics:** Industries: ${icp.firmographic.industries?.join(', ') || 'N/A'} | Size: ${icp.firmographic.companySizes?.join(', ') || 'N/A'} | Revenue: ${icp.firmographic.revenueRange || 'N/A'}`);
   345→    }
   346→    return parts.join('\n\n');
   347→  }
   348→
   349→  // Generic fallback
   350→  return `I've processed your request about "${userMessage.slice(0, 50)}". The research pipeline has completed. Check the results below for details, and let me know if you'd like me to take further action.`;
   351→}

   353→// ============================================================
   354→// Inner Orchestrator Implementation
   355→// ============================================================
   356→
   357→async function processWithOrchestratorInner(
   358→  userMessage: string,
   359→  context: ConversationContext | undefined,
   360→  forceIntent: UserIntent | undefined,
   361→  pipelineState: PipelineState,
   362→  startTime: number,
   363→  onEvent?: OrchestratorCallback,
   364→  resumeFrom?: PipelineCheckpoint,
   365→): Promise<{
   366→  message: AgentMessage;
   367→  updatedContext: ConversationContext;
   368→  suggestedActions: SuggestedAction[];
   369→  pipelineState: PipelineState;
   370→}> {
   371→
   372→  let updatedContext: ConversationContext = context || {
   373→    recentProspects: [], activeICP: null, lastIntent: null, lastPersona: null, userPreferences: {},
   374→  };
   375→
   376→  // ═══════════════════════════════════════════════════
   377→  // RESUME PATH: If a checkpoint is provided, skip
   378→  // the thinking phase and resume from the failed agent.
   379→  // ═══════════════════════════════════════════════════
   380→  let classification: IntentClassification;
   381→  let thinking: AgentThinking;
   382→
   383→  if (resumeFrom && resumeFrom.classifiedIntent) {
   384→    // Skip thinking — intent is already known from checkpoint
   385→    const personas: Record<UserIntent, AgentPersona> = {
   386→      research_company: 'scout', research_person: 'hound', research_url: 'scout',
   387→      analyze_market: 'analyst', analyze_competitors: 'analyst', build_icp: 'architect',
   388→      score_lead: 'judge', compose_outreach: 'scribe', refine_search: 'scout',
   389→      add_to_pipeline: 'navigator', clarify: 'navigator', converse: 'navigator',
   390→    };
   391→    classification = {
   392→      intent: resumeFrom.classifiedIntent,
   393→      persona: personas[resumeFrom.classifiedIntent],
   394→      confidence: 1.0,
   395→      reasoning: 'Resumed from checkpoint — intent already classified',
   396→      extractedEntities: { companyName: null, personName: null, url: null, industry: null, location: null },
   397→      clarifyingQuestion: null,
   398→    };
   399→    thinking = intentToThinking(classification);
   400→
   401→    // Mark already-completed agents in pipeline state
   402→    for (const completedAgent of resumeFrom.completedAgents) {
   403→      updateAgentState(pipelineState, completedAgent, {
   404→        status: 'completed',
   405→        currentStep: `Completed before resume`,
   406→        progress: 100,
   407→        completedAt: Date.now(),
   408→      }, onEvent);
   409→    }
   410→
   411→    // Emit pipeline_resumed event
   412→    emit(onEvent, { type: 'pipeline_resumed', data: {
   413→      resumedFrom: resumeFrom.failedAgent || 'unknown',
   414→      completedAgents: resumeFrom.completedAgents,
   415→      failedAgent: resumeFrom.failedAgent,
   416→    }});
   417→
   418→    sendCommMsg(pipelineState, 'atlas', 'all', 'broadcast',
   419→      `Pipeline resumed from checkpoint. Skipping ${resumeFrom.completedAgents.length} completed agents. Resuming from ${resumeFrom.failedAgent || 'next agent'}...`,
   420→      { completedAgents: resumeFrom.completedAgents, failedAgent: resumeFrom.failedAgent },
   421→      onEvent);
   422→
   423→    console.log(`[Orchestrator] Resuming from checkpoint: completed=[${resumeFrom.completedAgents.join(',')}], failed=${resumeFrom.failedAgent}`);
   424→  } else {
   425→    // ═══════════════════════════════════════════════════
   426→    // PHASE 1: THINK — Atlas classifies intent
   427→    // ═══════════════════════════════════════════════════
   428→    pipelineState.phase = 'thinking';
   429→    pipelineState.thinkStartTime = Date.now();
   430→    emit(onEvent, { type: 'thinking_start', data: { timestamp: pipelineState.thinkStartTime } });
   431→    emit(onEvent, { type: 'pipeline_progress', data: { phase: 'thinking', overallProgress: 5 } });
   432→
   433→    // Update Atlas status (use 'atlas' as the 8-agent key)
   434→    updateAgentState(pipelineState, 'atlas', {
   435→      status: 'thinking',
   436→      currentStep: 'Classifying intent',
   437→      progress: 0,
   438→      startedAt: Date.now(),
   439→    }, onEvent);
   440→
   441→    sendCommMsg(pipelineState, 'user', 'atlas', 'request',
   442→      `Classify this query: "${userMessage.slice(0, 100)}"`, undefined, onEvent);
   443→
   444→    // Run intent classification
   445→    if (forceIntent) {
   446→      const personas: Record<UserIntent, AgentPersona> = {
   447→        research_company: 'scout', research_person: 'hound', research_url: 'scout',
   448→        analyze_market: 'analyst', analyze_competitors: 'analyst', build_icp: 'architect',
   449→        score_lead: 'judge', compose_outreach: 'scribe', refine_search: 'scout',
   450→        add_to_pipeline: 'navigator', clarify: 'navigator', converse: 'navigator',
   451→      };
   452→      classification = {
   453→        intent: forceIntent,
   454→        persona: personas[forceIntent],
   455→        confidence: 1.0,
   456→        reasoning: 'Intent was explicitly specified',
   457→        extractedEntities: { companyName: null, personName: null, url: null, industry: null, location: null },
   458→        clarifyingQuestion: null,
   459→      };
   460→    } else {
   461→      classification = await classifyIntent(userMessage, context);
   462→    }
   463→
   464→    thinking = intentToThinking(classification);
   465→
   466→    // End thinking phase
   467→    const thinkEndTime = Date.now();
   468→    pipelineState.totalThinkTimeMs = thinkEndTime - (pipelineState.thinkStartTime || startTime);
   469→
   470→    // Update Atlas status to completed
   471→    updateAgentState(pipelineState, 'atlas', {
   472→      status: 'completed',
   473→      currentStep: `Intent: ${classification.intent} (${Math.round(classification.confidence * 100)}% confidence)`,
   474→      progress: 100,
   475→      completedAt: Date.now(),
   476→      thinkTimeMs: pipelineState.totalThinkTimeMs,
   477→    }, onEvent);
   478→
   479→    sendCommMsg(pipelineState, 'atlas', 'all', 'broadcast',
   480→      `Query classified as **${classification.intent}** with ${Math.round(classification.confidence * 100)}% confidence. Activating pipeline...`,
   481→      { intent: classification.intent, confidence: classification.confidence, persona: classification.persona },
   482→      onEvent);
   483→
   484→    emit(onEvent, { type: 'thinking_end', data: { totalMs: pipelineState.totalThinkTimeMs, classification } });
   485→  }
   486→
   487→  // ═══════════════════════════════════════════════════
   488→  // FAST PATH: For converse/clarify intents, skip the full
   489→  // pipeline and respond directly with a single LLM call.
   490→  // This avoids 2-3 minutes of unnecessary pipeline execution.
   491→  // ═══════════════════════════════════════════════════
   492→  if (classification.intent === 'converse' || classification.intent === 'clarify') {
   493→    let responseContent = '';
   494→    if (classification.intent === 'clarify' && classification.clarifyingQuestion) {
   495→      responseContent = classification.clarifyingQuestion;
   496→    } else {
   497→      try {
   498→        const contextHint = buildContextHint(updatedContext);
   499→        responseContent = await generateConversationResponse(
   500→          'navigator', classification.intent, userMessage,
   501→          contextHint || 'General conversation', updatedContext,
   502→        );
   503→      } catch {
   504→        // LLM failed — use a hardcoded fallback
   505→        if (classification.intent === 'converse') {
   506→          responseContent = "I'm here to help with B2B lead generation! You can ask me to:\n\n• **Research a company** — e.g., \"Tell me about Stripe\"\n• **Find a person** — e.g., \"Look up Patrick Collison\"\n• **Analyze a market** — e.g., \"What's the SaaS market in Berlin?\"\n• **Build an ICP** — e.g., \"Help me define my ideal customer\"\n• **Score a lead** — e.g., \"Is this a good lead?\"\n• **Compose outreach** — e.g., \"Write an email to Acme Corp\"";
   507→        } else {
   508→          responseContent = "I'd love to help! Could you tell me more about what you're looking for? For example:\n• Research a specific company\n• Find information about a person\n• Analyze a market or industry\n• Build an Ideal Customer Profile\n• Compose an outreach message";
   509→        }
   510→      }
   511→    }
   512→
   513→    if (!responseContent) {
   514→      responseContent = classification.intent === 'converse'
   515→        ? "I'm here to help with B2B lead generation! Ask me to research companies, find people, analyze markets, build ICPs, score leads, or compose outreach."
   516→        : "I'd love to help! Could you be more specific about what you're looking for?";
   517→    }
   518→
   519→    pipelineState.phase = 'complete';
   520→    emit(onEvent, { type: 'pipeline_progress', data: { phase: 'complete', overallProgress: 100 } });
   521→
   522→    // Mark echo as completed since we skip it
   523→    updateAgentState(pipelineState, 'echo', {
   524→      status: 'completed',
   525→      currentStep: 'Skipped — fast path',
   526→      progress: 100,
   527→      completedAt: Date.now(),
   528→    }, onEvent);
   529→
   530→    updatedContext.lastIntent = classification.intent;
   531→    updatedContext.lastPersona = classification.persona;
   532→
   533→    const agentMessage: AgentMessage = {
   534→      id: `agent-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`,
   535→      role: 'assistant',
   536→      content: responseContent,
   537→      timestamp: new Date(),
   538→      persona: classification.persona,
   539→      thinking,
   540→      actions: [{ type: classification.intent, label: 'Direct Response', status: 'completed', message: 'Fast-path response' }],
   541→    };
   542→
   543→    const suggestedActions = generateSuggestedActions(classification.intent, undefined, updatedContext);
   544→
   545→    sendCommMsg(pipelineState, 'atlas', 'user', 'response',
   546→      `Fast path: ${classification.intent} handled directly.`, undefined, onEvent);
   547→
   548→    console.log(`[Orchestrator] Fast-path "${userMessage.slice(0, 50)}" → intent=${classification.intent}, took=${Date.now() - startTime}ms`);
   549→
   550→    return { message: agentMessage, updatedContext, suggestedActions, pipelineState };
   551→  }
   552→
   553→  // ═══════════════════════════════════════════════════
   554→  // PHASE 2: EXECUTE — Run the 8-agent pipeline
   555→  // ═══════════════════════════════════════════════════
   556→  pipelineState.phase = 'executing';
   557→  emit(onEvent, { type: 'pipeline_progress', data: { phase: 'executing', overallProgress: 15 } });
   558→
   559→  let actions: AgentAction[] = [];
   560→  let prospectData: ProspectResult | undefined;
   561→  let icpData: ICPResult | undefined;
   562→  let outreachData: OutreachResult | undefined;
   563→  let marketData: MarketResult | undefined;
   564→  let scoreData: ScoreResult | undefined;
   565→  let responseContent = '';
   566→
   567→  // Initialize partial data from checkpoint if resuming
   568→  if (resumeFrom) {
   569→    if (resumeFrom.partialProspectData) prospectData = resumeFrom.partialProspectData as unknown as ProspectResult;
   570→    if (resumeFrom.partialIcpData) icpData = resumeFrom.partialIcpData as unknown as ICPResult;
   571→    if (resumeFrom.partialScoreData) scoreData = resumeFrom.partialScoreData as unknown as ScoreResult;
   572→    if (resumeFrom.partialOutreachData) outreachData = resumeFrom.partialOutreachData as unknown as OutreachResult;
   573→    if (resumeFrom.partialMarketData) marketData = resumeFrom.partialMarketData as unknown as MarketResult;
   574→    // Also restore context from partial data
   575→    if (prospectData) {
   576→      updatedContext.recentProspects = [...updatedContext.recentProspects, prospectData];
   577→    }
   578→  }
   579→
   580→  // Create a progress callback that bridges to the orchestrator events
   581→  const bridgeProgress: ProgressCallback = (event: string, data: unknown) => {
   582→    if (event === 'step_start') {
   583→      emit(onEvent, { type: 'step_start', data: data as { stepIndex: number; label: string; agent: AgentPersona; message: string } });
   584→    } else if (event === 'step_progress') {
   585→      emit(onEvent, { type: 'step_progress', data: data as { stepIndex: number; message: string; partialData?: Record<string, unknown> } });
   586→    } else if (event === 'step_complete') {
   587→      emit(onEvent, { type: 'step_complete', data: data as { stepIndex: number; status: 'completed' | 'failed'; message: string; partialData?: Record<string, unknown> } });
   588→    } else if (event === 'insight') {
   589→      emit(onEvent, { type: 'insight', data: data as { insight: InsightItem } });
   590→    }
   591→  };
   592→
   593→  // Determine which pipeline phases to execute based on intent
   594→  const relevantPhases = PIPELINE_PHASES.filter(phase => {
   595→    if (!phase.intentRequired) return true; // Always run (atlas, echo)
   596→    return phase.intentRequired.includes(classification.intent);
   597→  });
   598→
   599→  let stepIdx = 0;
   600→  const totalPhases = relevantPhases.length;
   601→
   602→  // Execute each relevant phase
   603→  for (const phase of relevantPhases) {
   604→    const agentKey = phase.agent; // 8-agent display name (atlas, scout, forge, etc.)
   605→
   606→    // ═══ SKIP COMPLETED AGENTS ON RESUME ═══
   607→    // If resuming from a checkpoint, skip agents that already completed
   608→    if (resumeFrom && resumeFrom.completedAgents.includes(phase.agent)) {
   609→      updateAgentState(pipelineState, agentKey, {
   610→        status: 'completed',
   611→        currentStep: `${phase.action} (completed before)`,
   612→        progress: 100,
   613→        completedAt: Date.now(),
   614→      }, onEvent);
   615→      stepIdx++;
   616→      continue;
   617→    }
   618→
   619→    // ═══ REDUCED COOLDOWN ═══
   620→    // Cooldown between agent phases to respect rate limit (concurrency=1).
   621→    // Reduced from 2-3.5s to 0.5-1.5s for faster pipeline execution.
   622→    if (stepIdx > 0) {
   623→      const isSearchPhase = phase.agent === 'scout';
   624→      const cooldownMs = isSearchPhase
   625→        ? 500 + Math.random() * 500    // 0.5-1s for search phases
   626→        : 1000 + Math.random() * 500;  // 1-1.5s for LLM phases
   627→
   628→      updateAgentState(pipelineState, agentKey, {
   629→        status: 'waiting',
   630→        currentStep: `Cooldown (${Math.round(cooldownMs / 1000)}s)`,
   631→        progress: 0,
   632→        startedAt: Date.now(),
   633→      }, onEvent);
   634→
   635→      sendCommMsg(pipelineState, 'atlas', agentKey, 'status',
   636→        `Cooldown: ${Math.round(cooldownMs / 1000)}s before starting`,
   637→        { cooldownMs, reason: 'rate_limit_buffer' },
   638→        onEvent);
   639→
   640→      emit(onEvent, { type: 'cooldown', data: { agent: agentKey, cooldownMs, reason: 'rate_limit_buffer' } });
   641→      emit(onEvent, { type: 'pipeline_progress', data: { phase: 'executing', overallProgress: Math.round(15 + (stepIdx / totalPhases) * 70 - 2) } });
   642→
   643→      await new Promise(r => setTimeout(r, cooldownMs));
   644→    }
   645→
   646→    // Update agent state to working
   647→    updateAgentState(pipelineState, agentKey, {
   648→      status: 'working',
   649→      currentStep: phase.action,
   650→      progress: 0,
   651→      startedAt: Date.now(),
   652→    }, onEvent);
   653→
   654→    // Atlas tells the agent what to do
   655→    if (phase.agent !== 'atlas') {
   656→      sendCommMsg(pipelineState, 'atlas', agentKey, 'request',
   657→        `[${agentKey.toUpperCase()}] ${phase.action} for query: "${userMessage.slice(0, 80)}"`,
   658→        { intent: classification.intent, phase: agentKey },
   659→        onEvent);
   660→    }
   661→
   662→    // Progress calculation
   663→    const baseProgress = 15 + (stepIdx / totalPhases) * 70;
   664→    emit(onEvent, { type: 'pipeline_progress', data: { phase: 'executing', overallProgress: Math.round(baseProgress) } });
   665→
   666→    // Execute the actual work based on the agent
   667→    try {
   668→      switch (phase.agent) {
   669→        case 'atlas':
   670→          // Already done (intent classification)
   671→          sendCommMsg(pipelineState, 'atlas', 'all', 'broadcast',
   672→            `Pipeline plan: ${relevantPhases.map(p => p.agent).join(' → ')}`, undefined, onEvent);
   673→          break;
   674→
   675→        case 'scout':
   676→          // Scout: Company/Person/URL research
   677→          if (['research_company', 'research_url', 'refine_search'].includes(classification.intent)) {
   678→            const companyName = classification.extractedEntities.companyName || userMessage.trim();
   679→            sendCommMsg(pipelineState, 'atlas', 'scout', 'handoff',
   680→              `Research target: "${companyName}"`, { target: companyName }, onEvent);
   681→
   682→            const result = await executeCompanyResearch(companyName, bridgeProgress);
   683→            actions = [...actions, ...result.steps];
   684→
   685→            if (result.prospect) {
   686→              prospectData = result.prospect;
   687→              updatedContext.recentProspects = [...updatedContext.recentProspects.slice(-4), result.prospect];
   688→
   689→              sendCommMsg(pipelineState, 'scout', 'atlas', 'response',
   690→                `Found data for "${result.prospect.companyName}" — ${result.prospect.dataCompleteness}% complete`,
   691→                { completeness: result.prospect.dataCompleteness, companyName: result.prospect.companyName },
   692→                onEvent);
   693→            } else {
   694→              sendCommMsg(pipelineState, 'scout', 'atlas', 'response',
   695→                'Limited data found. Continuing with partial results.', undefined, onEvent);
   696→            }
   697→          } else if (classification.intent === 'research_person') {
   698→            const personName = classification.extractedEntities.personName || userMessage.trim();
   699→            sendCommMsg(pipelineState, 'atlas', 'scout', 'handoff',
   700→              `Research person: "${personName}"`, { target: personName }, onEvent);
   701→
   702→            const result = await executePersonResearch(personName, bridgeProgress);
   703→            actions = [...actions, ...result.steps];
   704→            if (result.prospect) {
   705→              prospectData = result.prospect;
   706→              updatedContext.recentProspects = [...updatedContext.recentProspects.slice(-4), result.prospect];
   707→
   708→              sendCommMsg(pipelineState, 'scout', 'atlas', 'response',
   709→                `Found data for "${result.prospect.personName}" — ${result.prospect.dataCompleteness}% complete`,
   710→                { completeness: result.prospect.dataCompleteness, personName: result.prospect.personName },
   711→                onEvent);
   712→            }
   713→          }
   714→          break;
   715→
   716→        case 'forge':
   717→          // Forge: Data enrichment — deep crawl and gap fill
   718→          if (prospectData && prospectData.dataCompleteness < 80) {
   719→            sendCommMsg(pipelineState, 'scout', 'forge', 'handoff',
   720→              `Enriching data (currently ${prospectData.dataCompleteness}% complete)`,
   721→              { completeness: prospectData.dataCompleteness }, onEvent);
   722→
   723→            // Try additional enrichment searches
   724→            const enrichActions: AgentAction = {
   725→              type: 'research_company', label: 'Deep Enrichment', status: 'running',
   726→              message: `Enriching ${prospectData.companyName || 'prospect'} data...`,
   727→            };
   728→            actions.push(enrichActions);
   729→            const enrichIdx = actions.length - 1;
   730→
   731→            try {
   732→              // Try LinkedIn search for enrichment
   733→              if (prospectData.companyName) {
   734→                const liResult = await linkedInSearchCompanies(prospectData.companyName, 3);
   735→                if (liResult.success && liResult.data.length > 0) {
   736→                  const topResult = liResult.data[0];
   737→                  if (!prospectData.linkedinUrl && topResult.url) prospectData.linkedinUrl = topResult.url;
   738→                  if (!prospectData.description && topResult.headline) prospectData.description = topResult.headline;
   739→                }
   740→              }
   741→
   742→              // Try web search for missing data points
   743→              if (prospectData.companyName) {
   744→                const gapQueries: string[] = [];
   745→                if (!prospectData.ceoName) gapQueries.push(`"${prospectData.companyName}" CEO founder executive`);
   746→                if (!prospectData.revenueEstimate) gapQueries.push(`"${prospectData.companyName}" revenue funding valuation`);
   747→                if (!prospectData.employeeCount) gapQueries.push(`"${prospectData.companyName}" employees team size`);
   748→
   749→                if (gapQueries.length > 0) {
   750→                  const gapSearch = await exaSearch(gapQueries[0], 3);
   751→                  if (gapSearch.success && gapSearch.data.length > 0) {
   752→                    const allText = gapSearch.data.map(r => `${r.title} ${r.snippet}`).join(' ');
   753→
   754→                    // Simple regex extraction for common missing fields
   755→                    if (!prospectData.ceoName) {
   756→                      const ceoMatch = allText.match(/(?:CEO|Chief Executive|Founder)[:\s]+([A-Z][a-z]+ [A-Z][a-z]+)/);
   757→                      if (ceoMatch) prospectData.ceoName = ceoMatch[1];
   758→                    }
   759→                    if (!prospectData.revenueEstimate) {
   760→                      const revMatch = allText.match(/\$[\d.]+[BbMmKk]/);
   761→                      if (revMatch) prospectData.revenueEstimate = revMatch[0];
   762→                    }
   763→                    if (!prospectData.employeeCount) {
   764→                      const empMatch = allText.match(/(\d[\d,+]*)\s*(?:employees|team members|staff|people)/i);
   765→                      if (empMatch) prospectData.employeeCount = empMatch[1];
   766→                    }
   767→                  }
   768→                }
   769→              }
   770→
   771→              // Recalculate completeness
   772→              prospectData.dataCompleteness = calculateQuickCompleteness(prospectData);
   773→
   774→              actions[enrichIdx] = { ...enrichActions, status: 'completed', message: `Enrichment complete — ${prospectData.dataCompleteness}% data` };
   775→              sendCommMsg(pipelineState, 'forge', 'atlas', 'response',
   776→                `Enrichment brought data to ${prospectData.dataCompleteness}%`, undefined, onEvent);
   777→            } catch (e) {
   778→              actions[enrichIdx] = { ...enrichActions, status: 'completed', message: 'Enrichment partially completed' };
   779→            }
   780→
   781→            // ── Phase 1 Data Source Channels ──────────────────────────────
   782→            // Run OpenCorporates, yfinance, SEC EDGAR, News Worker, Geocoder,
   783→            // and PublicWWW in parallel to enrich the prospect with verified
   784→            // registry data, financial KPIs, news intent signals, and geo coords.
   785→            // PublicWWW is skipped here to respect its 60s rate limit — it's
   786→            // only called explicitly from the Scout phase for tech-stack discovery.
   787→            if (prospectData && (prospectData.companyName || prospectData.personName)) {
   788→              try {
   789→                const channelActions: AgentAction = {
   790→                  type: 'research_company',
   791→                  label: 'Data Source Channels',
   792→                  status: 'running',
   793→                  message: `Querying OpenCorporates, yfinance, SEC EDGAR, News, Geocoder...`,
   794→                };
   795→                actions.push(channelActions);
   796→                const chIdx = actions.length - 1;
   797→
   798→                const channelResult = await executeChannelEnrichment(prospectData, {
   799→                  onProgress: (label, message, status) => {
   800→                    if (status === 'running') {
   801→                      actions[chIdx] = { ...channelActions, status: 'running', message };
   802→                    } else if (status === 'completed') {
   803→                      actions[chIdx] = { ...channelActions, status: 'completed', message };
   804→                    }
   805→                  },
   806→                  skipPublicWww: true,  // 60s rate limit — call separately in Scout phase
   807→                });
   808→
   809→                // Replace prospectData with the enriched version
   810→                prospectData = channelResult.prospect;
   811→
   812→                const ocOk = channelResult.channelResults.openCorporates?.success ? '✓' : '✗';
   813→                const yfOk = channelResult.channelResults.yfinance?.success ? '✓' : '✗';
   814→                const edOk = channelResult.channelResults.edgar?.success ? '✓' : '✗';
   815→                const nwOk = channelResult.channelResults.news?.success ? '✓' : '✗';
   816→                const geOk = channelResult.channelResults.geocoder?.success ? '✓' : '✗';
   817→
   818→                actions[chIdx] = {
   819→                  ...channelActions,
   820→                  status: 'completed',
   821→                  message: `Channels: OC:${ocOk} YF:${yfOk} EDGAR:${edOk} News:${nwOk} Geo:${geOk} — ` +
   822→                           `${channelResult.personLeads.length} person leads, ` +
   823→                           `${Object.keys(channelResult.customKpis).length} KPIs`,
   824→                };
   825→
   826→                sendCommMsg(pipelineState, 'forge', 'atlas', 'response',
   827→                  `Channel enrichment: ${Object.keys(channelResult.customKpis).length} KPIs, ` +
   828→                  `${channelResult.personLeads.length} person leads`,
   829→                  { kpiCount: Object.keys(channelResult.customKpis).length,
   830→                    personLeadCount: channelResult.personLeads.length },
   831→                  onEvent);
   832→
   833→                // Recalculate completeness after channel enrichment
   834→                prospectData.dataCompleteness = calculateQuickCompleteness(prospectData);
   835→              } catch (e) {
   836→                // Non-fatal — the pipeline continues with whatever data we have
   837→                console.warn('[orchestrator] Channel enrichment failed:', e);
   838→              }
   839→            }
   840→          }
   841→          break;
   842→
   843→        case 'sage':
   844→          // Sage: Market/Competitive analysis
   845→          if (classification.intent === 'analyze_market') {
   846→            const result = await executeMarketAnalysis(userMessage);
   847→            actions = [...actions, ...result.steps];
   848→            if (result.market) {
   849→              marketData = result.market;
   850→              sendCommMsg(pipelineState, 'sage', 'atlas', 'response',
   851→                `Market analysis complete: ${result.market.keyFindings.length} findings, ${result.market.competitors.length} competitors`,
   852→                { findings: result.market.keyFindings.length, competitors: result.market.competitors.length },
   853→                onEvent);
   854→            }
   855→          } else if (classification.intent === 'analyze_competitors') {
   856→            const result = await executeCompetitiveAnalysis(userMessage);
   857→            actions = [...actions, ...result.steps];
   858→            if (result.market) {
   859→              marketData = result.market;
   860→              sendCommMsg(pipelineState, 'sage', 'atlas', 'response',
   861→                `Competitive analysis complete`,
   862→                undefined, onEvent);
   863→            }
   864→          }
   865→          break;
   866→
   867→        case 'judge':
   868→          // Judge: Lead scoring
   869→          {
   870→            const recentProspect = updatedContext.recentProspects[updatedContext.recentProspects.length - 1];
   871→            if (recentProspect) {
   872→              sendCommMsg(pipelineState, 'atlas', 'judge', 'request',
   873→                `Score ${recentProspect.companyName || recentProspect.personName} against ICP`,
   874→                { companyName: recentProspect.companyName }, onEvent);
   875→
   876→              // Refresh news intent signals right before scoring — this gives the
   877→              // Judge the freshest possible view of recent news/market activity.
   878→              try {
   879→                const targetName = recentProspect.companyName || recentProspect.personName || '';
   880→                if (targetName) {
   881→                  const newsRefresh = await refreshNewsIntent(targetName);
   882→                  if (newsRefresh.intent && newsRefresh.intent.success) {
   883→                    const kpiKeys = Object.keys(newsRefresh.kpis);
   884→                    if (kpiKeys.length > 0) {
   885→                      // Merge fresh news KPIs into prospect's customKpis
   886→                      const existingKpis = recentProspect.customKpis
   887→                        ? JSON.parse(recentProspect.customKpis)
   888→                        : {};
   889→                      const merged = { ...existingKpis, ...newsRefresh.kpis };
   890→                      recentProspect.customKpis = JSON.stringify(merged);
   891→
   892→                      sendCommMsg(pipelineState, 'judge', 'atlas', 'response',
   893→                        `Refreshed news intent: ${newsRefresh.intent.mentionCount30d} mentions, ` +
   894→                        `${kpiKeys.length} signals (sentiment: ${newsRefresh.intent.sentiment.overall})`,
   895→                        { mentions: newsRefresh.intent.mentionCount30d, sentiment: newsRefresh.intent.sentiment.overall },
   896→                        onEvent);
   897→                    }
   898→                  }
   899→                }
   900→              } catch (e) {
   901→                console.warn('[orchestrator] News refresh in Judge phase failed:', e);
   902→              }
   903→
   904→              const result = await executeLeadScoring(recentProspect, updatedContext.activeICP);
   905→              actions = [...actions, ...result.steps];
   906→              if (result.score) {
   907→                scoreData = result.score;
   908→                sendCommMsg(pipelineState, 'judge', 'atlas', 'response',
   909→                  `Lead score: ${result.score.overallScore}/100 (${result.score.tier})`,
   910→                  { score: result.score.overallScore, tier: result.score.tier },
   911→                  onEvent);
   912→              }
   913→            } else if (classification.intent === 'score_lead') {
   914→              responseContent = "I don't have a prospect to score yet. Please research a company or person first, then I can score them against your ICP.";
   915→            }
   916→          }
   917→          break;
   918→
   919→        case 'bard':
   920→          // Bard: Outreach composition
   921→          {
   922→            const recentProspect = updatedContext.recentProspects[updatedContext.recentProspects.length - 1];
   923→            if (recentProspect) {
   924→              const channel = userMessage.toLowerCase().includes('linkedin') ? 'linkedin' : 'email';
   925→              sendCommMsg(pipelineState, 'atlas', 'bard', 'request',
   926→                `Compose ${channel} outreach for ${recentProspect.companyName || recentProspect.personName}`,
   927→                { channel, target: recentProspect.companyName }, onEvent);
   928→
   929→              const result = await executeOutreachComposition(recentProspect, channel);
   930→              actions = [...actions, ...result.steps];
   931→              if (result.outreach) {
   932→                outreachData = result.outreach;
   933→                sendCommMsg(pipelineState, 'bard', 'atlas', 'response',
   934→                  `${channel} outreach composed with ${result.outreach.personalizationHooks?.length || 0} personalization hooks`,
   935→                  { channel, hooks: result.outreach.personalizationHooks?.length },
   936→                  onEvent);
   937→              }
   938→            } else if (classification.intent === 'compose_outreach') {
   939→              const target = userMessage.replace(/^(?:write|compose|draft|send|email|reach out|craft)\s*(?:an?\s*)?(?:email|message|outreach|connection)?\s*(?:to\s*)?/i, '').trim();
   940→              if (target.length > 2) {
   941→                responseContent = `I need to research "${target}" first to write a personalized message. Would you like me to research them?`;
   942→              } else {
   943→                responseContent = "I need a prospect to compose outreach for. Please research a company or person first.";
   944→              }
   945→            }
   946→          }
   947→          break;
   948→
   949→        case 'flow':
   950→          // Flow: Pipeline management — saves session context and manages pipeline state
   951→          if (classification.intent === 'build_icp') {
   952→            const result = await executeICPBuilding(userMessage, updatedContext.activeICP);
   953→            actions = [...actions, ...result.steps];
   954→            if (result.icp) {
   955→              icpData = result.icp;
   956→              updatedContext.activeICP = result.icp;
   957→              responseContent = result.response;
   958→            }
   959→          } else if (classification.intent === 'add_to_pipeline') {
   960→            const recentProspect = updatedContext.recentProspects[updatedContext.recentProspects.length - 1];
   961→            const name = recentProspect?.companyName || recentProspect?.personName || 'this prospect';
   962→            responseContent = `Click the "Add to Leads" button below to add ${name} to your lead pipeline.`;
   963→          } else {
   964→            // Flow manages pipeline context for all other intents
   965→            const prospectCount = updatedContext.recentProspects.length;
   966→            const hasICP = !!updatedContext.activeICP;
   967→            const hasScoredLeads = updatedContext.recentProspects.some(p => (p as unknown as Record<string, unknown>).leadScore !== undefined);
   968→            let flowMsg = `Pipeline context: ${prospectCount} prospect(s) discovered`;
   969→            if (hasICP) flowMsg += `, ICP active`;
   970→            if (hasScoredLeads) flowMsg += `, leads scored`;
   971→            flowMsg += '. ';
   972→            if (prospectCount > 0 && !hasICP) {
   973→              flowMsg += 'Suggest building an ICP to improve qualification.';
   974→            } else if (hasScoredLeads) {
   975→              flowMsg += 'Recommend composing outreach for scored leads.';
   976→            } else if (prospectCount > 0) {
   977→              flowMsg += 'Recommend scoring prospects against ICP.';
   978→            }
   979→            sendCommMsg(pipelineState, 'flow', 'atlas', 'response', flowMsg, undefined, onEvent);
   980→          }
   981→          break;
   982→
   983→        case 'echo':
   984→          // Echo: Insights and reporting — compiles final insights from all agents
   985→          {
   986→            const echoInsights: string[] = [];
   987→            if (prospectData) {
   988→              echoInsights.push(`Prospect research completed for ${prospectData.companyName || prospectData.personName || 'target'}`);
   989→              if (prospectData.employeeCount || prospectData.revenueEstimate) {

  1158→      emit(onEvent, { type: 'insight', data: { insight } });

  1160→  }
  1161→
  1162→  // Generate navigation suggestions
  1163→  const navigation = generateNavigationSuggestions(classification.intent, prospectData, icpData, scoreData, outreachData, marketData);

  1170→  const agentMessage: AgentMessage = {

  1173→    content: responseContent,

  1181→    marketData,

  1183→    insights: insights.length > 0 ? insights : undefined,
  1184→    navigation: navigation.length > 0 ? navigation : undefined,
  1185→  };
  1186→
  1187→  // Generate suggested actions
  1188→  const suggestedActions = generateSuggestedActions(classification.intent, prospectData, updatedContext);
  1189→
  1190→  // Final comm message
  1191→  sendCommMsg(pipelineState, 'atlas', 'user', 'response',
  1192→    `Pipeline complete. ${actions.length} steps executed, ${actions.filter(a => a.status === 'completed').length} succeeded.`,
  1193→    { steps: actions.length, completed: actions.filter(a => a.status === 'completed').length },
  1194→    onEvent);
  1195→
  1196→  console.log(`[Orchestrator] Processed "${userMessage.slice(0, 50)}" → intent=${classification.intent}, took=${Date.now() - startTime}ms`);
  1197→
  1198→  return {
  1199→    message: agentMessage,
  1200→    updatedContext,
  1201→    suggestedActions,
  1202→    pipelineState,
  1203→  };
  1204→}

  1210→function calculateQuickCompleteness(prospect: ProspectResult): number {
  1211→  const fields: Array<{ key: keyof ProspectResult; weight: number }> = [
  1212→    { key: 'companyName', weight: 10 },
  1213→    { key: 'website', weight: 8 },
  1214→    { key: 'industry', weight: 8 },
  1215→    { key: 'description', weight: 6 },
  1216→    { key: 'ceoName', weight: 7 },
  1217→    { key: 'employeeCount', weight: 6 },
  1218→    { key: 'revenueEstimate', weight: 6 },
  1219→    { key: 'generalEmail', weight: 6 },
  1220→    { key: 'phoneMain', weight: 5 },
  1221→    { key: 'linkedinUrl', weight: 5 },
  1222→    { key: 'city', weight: 4 },
  1223→    { key: 'country', weight: 4 },
  1224→    { key: 'foundingYear', weight: 3 },
  1225→    { key: 'twitterHandle', weight: 3 },
  1226→    { key: 'fundingInfo', weight: 4 },
  1227→    { key: 'techStack', weight: 5 },
  1228→    { key: 'recentNews', weight: 5 },
  1229→    { key: 'productsServices', weight: 5 },
  1230→  ];
  1231→
  1232→  let total = 0;
  1233→  let maxTotal = 0;
  1234→  for (const { key, weight } of fields) {
  1235→    maxTotal += weight;
  1236→    const val = prospect[key];
  1237→    if (val !== null && val !== undefined && val !== '' && !(Array.isArray(val) && val.length === 0)) {
  1238→      total += weight;
  1239→    }
  1240→  }
  1241→  return Math.round((total / maxTotal) * 100);
  1242→}

  1281→    parts.push('Would you like me to score this lead, compose outreach, or find similar companies?');

  1283→  return parts.join('\n\n');
  1284→}

  1397→        actions.push({ label: 'Add to Leads', prompt: 'Add this prospect to my leads', icon: 'Plus' });

  1399→          actions.push({ label: 'Find Similar', prompt: `Find similar companies in ${prospect.industry}`, icon: 'Search' });
  1400→        }
  1401→      }
  1402→      break;
  1403→
  1404→    case 'analyze_market':
  1405→    case 'analyze_competitors':
  1406→      actions.push({ label: 'Research Top Company', prompt: 'Research the top company from this analysis', icon: 'Building2' });
  1407→      if (!hasICP) actions.push({ label: 'Build an ICP', prompt: 'Build an ICP for this market', icon: 'Target' });
  1408→      break;
  1409→
  1410→    case 'score_lead':
  1411→      if (prospect) {
  1412→        actions.push({ label: 'Compose Outreach', prompt: `Write an email to ${prospectName}`, icon: 'Mail' });
  1413→        actions.push({ label: 'Add to Leads', prompt: 'Add this prospect to my leads', icon: 'Plus' });

  1416→
  1417→    case 'compose_outreach':
  1418→      if (prospect) {
  1419→        actions.push({ label: 'Add to Leads', prompt: 'Add this prospect to my leads', icon: 'Plus' });
  1420→        actions.push({ label: 'Find Similar', prompt: 'Find similar companies', icon: 'Search' });
  1421→      }
  1422→      break;
  1423→
  1424→    default:

  1428→  }

  1430→  return actions.slice(0, 4);
