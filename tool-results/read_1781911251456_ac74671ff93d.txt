   153→
   154→export async function processWithOrchestrator(
   155→  userMessage: string,
   156→  context?: ConversationContext,
   157→  forceIntent?: UserIntent,
   158→  onEvent?: OrchestratorCallback,
   159→  resumeFrom?: PipelineCheckpoint,
   160→): Promise<{
   161→  message: AgentMessage;
   162→  updatedContext: ConversationContext;
   163→  suggestedActions: SuggestedAction[];
   164→  pipelineState: PipelineState;
   165→}> {
   166→  const pipelineState = createInitialPipelineState();
   167→  const startTime = Date.now();
   168→
   169→  try {
   170→    return await processWithOrchestratorInner(userMessage, context, forceIntent, pipelineState, startTime, onEvent, resumeFrom);
   171→  } catch (error) {
   172→    console.error('[Orchestrator] FATAL:', error);
   173→
   174→    const errorMsg = error instanceof Error ? error.message : 'Unknown error';
   175→    pipelineState.phase = 'error';
   176→
   177→    // ─── GRACEFUL DEGRADE: Build a fallback response from user-supplied data ───
   178→    // Even if the entire pipeline failed, we still want to:
   179→    //   1. Parse the user's query for any structured data they provided
   180→    //   2. Build a minimal ProspectResult from that data
   181→    //   3. Render a meaningful response instead of just "Pipeline Error"
   182→    //
   183→    // This is the fix for the "Both stream and chat API failed" failure mode
   184→    // where users saw all "Discovering..." placeholders with no data.
   185→    let fallbackProspect: ProspectResult | undefined;
   186→    let fallbackContent: string;
   187→
   188→    try {
   189→      const { parseQuery } = await import('./query-parser');
   190→      const parsed = parseQuery(userMessage);
   191→
   192→      if (parsed.signalsProvided > 0 && parsed.prepopulatedProspect) {
   193→        // Build a real ProspectResult from pre-populated data
   194→        fallbackProspect = {
   195→          ...({
   196→            queryType: parsed.guessedIntent === 'research_person' ? 'person' : 'company',
   197→            query: userMessage,
   198→            companyName: null, legalName: null, website: null, industry: null, subIndustry: null, description: null,
   199→            hqAddress: null, city: null, stateProvince: null, country: null, postalCode: null,
   200→            phoneMain: null, generalEmail: null, supportEmail: null,
   201→            ceoName: null, ceoEmail: null, keyContactName: null, keyContactTitle: null, keyContactEmail: null,
   202→            employeeCount: null, revenueEstimate: null, foundingYear: null, ownershipType: null,
   203→            linkedinUrl: null, twitterHandle: null, facebookPage: null, techStack: [],
   204→            boardMembers: [], recentNews: [], productsServices: [], partners: [], fundingInfo: null,
   205→            personName: null, personTitle: null, personCompany: null, personEmail: null,
   206→            personPhone: null, personLinkedin: null, personBio: null,
   207→            sources: [], dataCompleteness: 0,
   208→          } as ProspectResult),
   209→          ...(parsed.prepopulatedProspect as Record<string, unknown>),
   210→        } as unknown as ProspectResult;
   211→
   212→        // Calculate completeness based on what we have
   213→        fallbackProspect.dataCompleteness = Math.max(
   214→          fallbackProspect.dataCompleteness || 0,
   215→          Math.min(70, parsed.signalsProvided * 8), // cap at 70% since research is incomplete
   216→        );
   217→
   218→        // Build a user-friendly fallback message
   219→        const filledFields: string[] = [];
   220→        if (fallbackProspect.personName) filledFields.push(`**Name:** ${fallbackProspect.personName}`);
   221→        if (fallbackProspect.personTitle) filledFields.push(`**Title:** ${fallbackProspect.personTitle}`);
   222→        if (fallbackProspect.personCompany || fallbackProspect.companyName) {
   223→          filledFields.push(`**Company:** ${fallbackProspect.personCompany || fallbackProspect.companyName}`);
   224→        }
   225→        if (fallbackProspect.personEmail || fallbackProspect.generalEmail) {
   226→          filledFields.push(`**Email:** ${fallbackProspect.personEmail || fallbackProspect.generalEmail}`);
   227→        }
   228→        if (fallbackProspect.personLinkedin) filledFields.push(`**LinkedIn:** ${fallbackProspect.personLinkedin}`);
   229→        if (fallbackProspect.city || fallbackProspect.country) {
   230→          filledFields.push(`**Location:** ${[fallbackProspect.city, fallbackProspect.country].filter(Boolean).join(', ')}`);
   231→        }
   232→        if (fallbackProspect.industry) filledFields.push(`**Industry:** ${fallbackProspect.industry}`);
   233→        if (fallbackProspect.personBio) filledFields.push(`**Bio:** ${fallbackProspect.personBio.slice(0, 200)}${fallbackProspect.personBio.length > 200 ? '...' : ''}`);
   234→
   235→        fallbackContent = `I extracted the following information from your query. However, my external research services encountered an error (${errorMsg.slice(0, 100)}), so I couldn't enrich this with additional web data. Here's what you provided:\n\n${filledFields.join('\n')}\n\nYou can try running this query again in a moment, or use the data above as-is.`;
   236→      } else {
   237→        fallbackContent = `I encountered an error while processing your request. Let me try a different approach.\n\n**Error:** ${errorMsg.slice(0, 150)}\n\nYou can try:\n• Rephrasing your query (e.g., "Research Stripe")\n• Asking a more specific question`;
   238→      }
   239→    } catch (parseErr) {
   240→      console.error('[Orchestrator] Fallback parser also failed:', parseErr);
   241→      fallbackContent = `I encountered an error while processing your request. Let me try a different approach.\n\n**Error:** ${errorMsg.slice(0, 150)}\n\nYou can try:\n• Rephrasing your query (e.g., "Research Stripe")\n• Asking a more specific question`;
   242→    }
   243→
   244→    return {
   245→      message: {
   246→        id: `agent-fallback-${Date.now()}`,
   247→        role: 'assistant',
   248→        content: fallbackContent,
   249→        timestamp: new Date(),
   250→        persona: 'navigator',
   251→        thinking: {
   252→          persona: 'navigator',
   253→          intent: 'converse',
   254→          reasoning: `Fallback after pipeline error: ${errorMsg.slice(0, 100)}`,
   255→          plan: ['Error recovery', 'Render user-supplied data'],
   256→          confidence: 0.1,
   257→        },
   258→        actions: [{ type: 'converse', label: 'Error', status: 'failed', message: errorMsg.slice(0, 100) }],
   259→        prospectData: fallbackProspect,
   260→      },
   261→      updatedContext: context || { recentProspects: [], activeICP: null, lastIntent: null, lastPersona: null, userPreferences: {} },
   262→      suggestedActions: [
   263→        { label: 'Try Again', prompt: userMessage, icon: 'RefreshCw' },
   264→        { label: 'Help', prompt: 'What can you do?', icon: 'Lightbulb' },
   265→      ],
   266→      pipelineState,
   267→    };
   268→  }
   269→}

   281→// Uses 8-agent display names (atlas, scout, forge, etc.) for

   284→
   285→function sendCommMsg(
   286→  pipelineState: PipelineState,
   287→  from: string,
   288→  to: string,
   289→  type: AgentCommMessage['type'],
   290→  content: string,
   291→  data?: Record<string, unknown>,
   292→  onEvent?: OrchestratorCallback,
   293→): void {
   294→  const msg: AgentCommMessage = {
   295→    id: `comm-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`,
   296→    from, to, type, content, data, timestamp: Date.now(),
   297→  };
   298→  pipelineState.commLog.push(msg);
   299→  emit(onEvent, { type: 'agent_comm', data: msg });
   300→}

   302→// ============================================================
   303→// Helper: update agent state
   304→// Uses the 8-agent display name as the key in pipelineState.agents,
   305→// and sets the persona field from AGENT_8_MAP for backward compat.
   306→// ============================================================
   307→
   308→function updateAgentState(
   309→  pipelineState: PipelineState,
   310→  agentKey: string,  // 8-agent display name (atlas, scout, forge, etc.)
   311→  update: Partial<AgentState>,
   312→  onEvent?: OrchestratorCallback,
   313→): void {
   314→  // Ensure persona is set from the mapping if not already provided
   315→  if (!update.persona) {
   316→    update.persona = AGENT_8_MAP[agentKey] || 'navigator';
   317→  }
   318→  const existing = pipelineState.agents[agentKey];
   319→  if (existing) {
   320→    pipelineState.agents[agentKey] = { ...existing, ...update };
   321→  } else {
   322→    // Create the entry if it doesn't exist (safety net)
   323→    pipelineState.agents[agentKey] = {
   324→      persona: update.persona,
   325→      status: 'idle',
   326→      currentStep: '',
   327→      progress: 0,
   328→      startedAt: null,
   329→      completedAt: null,
   330→      thinkTimeMs: null,
   331→      ...update,
   332→    };
   333→  }
   334→  emit(onEvent, { type: 'agent_status', data: { agent: agentKey, state: pipelineState.agents[agentKey] } });
   335→}

   337→// ============================================================
   338→// Helper: build rich fallback response from structured data
   339→// without needing an LLM call. Covers all intent types.
   340→// ============================================================
   341→
   342→function buildRichFallbackResponse(params: {
   343→  intent: UserIntent;
   344→  prospect?: ProspectResult;
   345→  icp?: ICPResult;
   346→  market?: MarketResult;
   347→  score?: ScoreResult;
   348→  outreach?: OutreachResult;
   349→  userMessage: string;
   350→}): string {
   351→  const { intent, prospect, icp, market, score, outreach, userMessage } = params;
   352→
   353→  // Prospect-based responses
   354→  if (prospect) {
   355→    return buildFallbackResponse(prospect, intent);
   356→  }
   357→
   358→  // Market analysis fallback
   359→  if (market) {
   360→    const parts: string[] = ['## Market Analysis'];
   361→    if (market.summary) parts.push(market.summary);
   362→    if (market.keyFindings?.length) {
   363→      parts.push('\n**Key Findings:**');
   364→      market.keyFindings.slice(0, 5).forEach(f => parts.push(`- ${f}`));
   365→    }
   366→    if (market.competitors?.length) {
   367→      parts.push('\n**Competitors:**');
   368→      market.competitors.slice(0, 5).forEach(c => parts.push(`- **${c.name}**: ${c.description?.slice(0, 80) || 'No description'}`));
   369→    }
   370→    if (market.trends?.length) {
   371→      parts.push('\n**Trends:**');
   372→      market.trends.slice(0, 3).forEach(t => parts.push(`- ${t}`));
   373→    }
   374→    return parts.join('\n\n');
   375→  }
   376→
   377→  // Score fallback
   378→  if (score) {
   379→    const parts: string[] = ['## Lead Score'];
   380→    parts.push(`**Overall Score:** ${score.overallScore}/100 — **${score.tier.toUpperCase()}** tier`);
   381→    parts.push(`**Recommendation:** ${score.recommendation}`);
   382→    if (score.dimensions) {
   383→      const dims = Object.entries(score.dimensions) as [string, { score: number; reasoning: string }][];
   384→      parts.push('\n**Dimension Breakdown:**');
   385→      dims.forEach(([key, val]) => {
   386→        parts.push(`- **${key}**: ${val.score}/100 — ${val.reasoning?.slice(0, 60) || ''}`);
   387→      });
   388→    }
   389→    return parts.join('\n\n');
   390→  }
   391→
   392→  // Outreach fallback
   393→  if (outreach) {
   394→    const parts: string[] = ['## Outreach Message'];
   395→    if (outreach.subject) parts.push(`**Subject:** ${outreach.subject}`);
   396→    parts.push(`**Channel:** ${outreach.channel}`);
   397→    parts.push(`**Tone:** ${outreach.tone}`);
   398→    parts.push(`\n${outreach.body}`);
   399→    if (outreach.personalizationHooks?.length) {
   400→      parts.push('\n**Personalization Hooks:**');
   401→      outreach.personalizationHooks.forEach(h => parts.push(`- ${h}`));
   402→    }
   403→    return parts.join('\n\n');
   404→  }
   405→
   406→  // ICP fallback
   407→  if (icp) {
   408→    const parts: string[] = ['## Ideal Customer Profile'];
   409→    parts.push(`**${icp.name}**`);
   410→    if (icp.description) parts.push(icp.description);
   411→    if (icp.firmographic) {
   412→      parts.push(`\n**Firmographics:** Industries: ${icp.firmographic.industries?.join(', ') || 'N/A'} | Size: ${icp.firmographic.companySizes?.join(', ') || 'N/A'} | Revenue: ${icp.firmographic.revenueRange || 'N/A'}`);
   413→    }
   414→    return parts.join('\n\n');
   415→  }
   416→
   417→  // Generic fallback
   418→  return `I've processed your request about "${userMessage.slice(0, 50)}". The research pipeline has completed. Check the results below for details, and let me know if you'd like me to take further action.`;
   419→}

   421→// ============================================================
   422→// Inner Orchestrator Implementation
   423→// ============================================================
   424→
   425→async function processWithOrchestratorInner(
   426→  userMessage: string,
   427→  context: ConversationContext | undefined,
   428→  forceIntent: UserIntent | undefined,
   429→  pipelineState: PipelineState,
   430→  startTime: number,
   431→  onEvent?: OrchestratorCallback,
   432→  resumeFrom?: PipelineCheckpoint,
   433→): Promise<{
   434→  message: AgentMessage;
   435→  updatedContext: ConversationContext;
   436→  suggestedActions: SuggestedAction[];
   437→  pipelineState: PipelineState;
   438→}> {
   439→
   440→  let updatedContext: ConversationContext = context || {
   441→    recentProspects: [], activeICP: null, lastIntent: null, lastPersona: null, userPreferences: {},
   442→  };
   443→
   444→  // ═══════════════════════════════════════════════════
   445→  // RESUME PATH: If a checkpoint is provided, skip
   446→  // the thinking phase and resume from the failed agent.
   447→  // ═══════════════════════════════════════════════════
   448→  let classification: IntentClassification;
   449→  let thinking: AgentThinking;
   450→
   451→  if (resumeFrom && resumeFrom.classifiedIntent) {
   452→    // Skip thinking — intent is already known from checkpoint
   453→    const personas: Record<UserIntent, AgentPersona> = {
   454→      research_company: 'scout', research_person: 'hound', research_url: 'scout',
   455→      analyze_market: 'analyst', analyze_competitors: 'analyst', build_icp: 'architect',
   456→      score_lead: 'judge', compose_outreach: 'scribe', refine_search: 'scout',
   457→      add_to_pipeline: 'navigator', clarify: 'navigator', converse: 'navigator',
   458→    };
   459→    classification = {
   460→      intent: resumeFrom.classifiedIntent,
   461→      persona: personas[resumeFrom.classifiedIntent],
   462→      confidence: 1.0,
   463→      reasoning: 'Resumed from checkpoint — intent already classified',
   464→      extractedEntities: { companyName: null, personName: null, url: null, industry: null, location: null },
   465→      clarifyingQuestion: null,
   466→    };
   467→    thinking = intentToThinking(classification);
   468→
   469→    // Mark already-completed agents in pipeline state
   470→    for (const completedAgent of resumeFrom.completedAgents) {
   471→      updateAgentState(pipelineState, completedAgent, {
   472→        status: 'completed',
   473→        currentStep: `Completed before resume`,
   474→        progress: 100,
   475→        completedAt: Date.now(),
   476→      }, onEvent);
   477→    }
   478→
   479→    // Emit pipeline_resumed event
   480→    emit(onEvent, { type: 'pipeline_resumed', data: {
   481→      resumedFrom: resumeFrom.failedAgent || 'unknown',
   482→      completedAgents: resumeFrom.completedAgents,
   483→      failedAgent: resumeFrom.failedAgent,
   484→    }});
   485→
   486→    sendCommMsg(pipelineState, 'atlas', 'all', 'broadcast',
   487→      `Pipeline resumed from checkpoint. Skipping ${resumeFrom.completedAgents.length} completed agents. Resuming from ${resumeFrom.failedAgent || 'next agent'}...`,
   488→      { completedAgents: resumeFrom.completedAgents, failedAgent: resumeFrom.failedAgent },
   489→      onEvent);
   490→
   491→    console.log(`[Orchestrator] Resuming from checkpoint: completed=[${resumeFrom.completedAgents.join(',')}], failed=${resumeFrom.failedAgent}`);
   492→  } else {
   493→    // ═══════════════════════════════════════════════════
   494→    // PHASE 1: THINK — Atlas classifies intent
   495→    // ═══════════════════════════════════════════════════
   496→    pipelineState.phase = 'thinking';
   497→    pipelineState.thinkStartTime = Date.now();
   498→    emit(onEvent, { type: 'thinking_start', data: { timestamp: pipelineState.thinkStartTime } });
   499→    emit(onEvent, { type: 'pipeline_progress', data: { phase: 'thinking', overallProgress: 5 } });
   500→
   501→    // Update Atlas status (use 'atlas' as the 8-agent key)
   502→    updateAgentState(pipelineState, 'atlas', {
   503→      status: 'thinking',
   504→      currentStep: 'Classifying intent',
   505→      progress: 0,
   506→      startedAt: Date.now(),
   507→    }, onEvent);
   508→
   509→    sendCommMsg(pipelineState, 'user', 'atlas', 'request',
   510→      `Classify this query: "${userMessage.slice(0, 100)}"`, undefined, onEvent);
   511→
   512→    // Run intent classification
   513→    if (forceIntent) {
   514→      const personas: Record<UserIntent, AgentPersona> = {
   515→        research_company: 'scout', research_person: 'hound', research_url: 'scout',
   516→        analyze_market: 'analyst', analyze_competitors: 'analyst', build_icp: 'architect',
   517→        score_lead: 'judge', compose_outreach: 'scribe', refine_search: 'scout',
   518→        add_to_pipeline: 'navigator', clarify: 'navigator', converse: 'navigator',
   519→      };
   520→      classification = {
   521→        intent: forceIntent,
   522→        persona: personas[forceIntent],
   523→        confidence: 1.0,
   524→        reasoning: 'Intent was explicitly specified',
   525→        extractedEntities: { companyName: null, personName: null, url: null, industry: null, location: null },
   526→        clarifyingQuestion: null,
   527→      };
   528→    } else {
   529→      classification = await classifyIntent(userMessage, context);
   530→    }
   531→
   532→    thinking = intentToThinking(classification);
   533→
   534→    // End thinking phase
   535→    const thinkEndTime = Date.now();
   536→    pipelineState.totalThinkTimeMs = thinkEndTime - (pipelineState.thinkStartTime || startTime);
   537→
   538→    // Update Atlas status to completed
   539→    updateAgentState(pipelineState, 'atlas', {
   540→      status: 'completed',
   541→      currentStep: `Intent: ${classification.intent} (${Math.round(classification.confidence * 100)}% confidence)`,
   542→      progress: 100,
   543→      completedAt: Date.now(),
   544→      thinkTimeMs: pipelineState.totalThinkTimeMs,
   545→    }, onEvent);
   546→
   547→    sendCommMsg(pipelineState, 'atlas', 'all', 'broadcast',
   548→      `Query classified as **${classification.intent}** with ${Math.round(classification.confidence * 100)}% confidence. Activating pipeline...`,
   549→      { intent: classification.intent, confidence: classification.confidence, persona: classification.persona },
   550→      onEvent);
   551→
   552→    emit(onEvent, { type: 'thinking_end', data: { totalMs: pipelineState.totalThinkTimeMs, classification } });
   553→  }
   554→
   555→  // ═══════════════════════════════════════════════════
   556→  // FAST PATH: For converse/clarify intents, skip the full
   557→  // pipeline and respond directly with a single LLM call.
   558→  // This avoids 2-3 minutes of unnecessary pipeline execution.
   559→  // ═══════════════════════════════════════════════════
   560→  if (classification.intent === 'converse' || classification.intent === 'clarify') {
   561→    let responseContent = '';
   562→    if (classification.intent === 'clarify' && classification.clarifyingQuestion) {
   563→      responseContent = classification.clarifyingQuestion;
   564→    } else {
   565→      try {
   566→        const contextHint = buildContextHint(updatedContext);
   567→        responseContent = await generateConversationResponse(
   568→          'navigator', classification.intent, userMessage,
   569→          contextHint || 'General conversation', updatedContext,
   570→        );
   571→      } catch {
   572→        // LLM failed — use a hardcoded fallback
   573→        if (classification.intent === 'converse') {
   574→          responseContent = "I'm here to help with B2B lead generation! You can ask me to:\n\n• **Research a company** — e.g., \"Tell me about Stripe\"\n• **Find a person** — e.g., \"Look up Patrick Collison\"\n• **Analyze a market** — e.g., \"What's the SaaS market in Berlin?\"\n• **Build an ICP** — e.g., \"Help me define my ideal customer\"\n• **Score a lead** — e.g., \"Is this a good lead?\"\n• **Compose outreach** — e.g., \"Write an email to Acme Corp\"";
   575→        } else {
   576→          responseContent = "I'd love to help! Could you tell me more about what you're looking for? For example:\n• Research a specific company\n• Find information about a person\n• Analyze a market or industry\n• Build an Ideal Customer Profile\n• Compose an outreach message";
   577→        }
   578→      }
   579→    }
   580→
   581→    if (!responseContent) {
   582→      responseContent = classification.intent === 'converse'
   583→        ? "I'm here to help with B2B lead generation! Ask me to research companies, find people, analyze markets, build ICPs, score leads, or compose outreach."
   584→        : "I'd love to help! Could you be more specific about what you're looking for?";
   585→    }
   586→
   587→    pipelineState.phase = 'complete';
   588→    emit(onEvent, { type: 'pipeline_progress', data: { phase: 'complete', overallProgress: 100 } });
   589→
   590→    // Mark echo as completed since we skip it
   591→    updateAgentState(pipelineState, 'echo', {
   592→      status: 'completed',
   593→      currentStep: 'Skipped — fast path',
   594→      progress: 100,
   595→      completedAt: Date.now(),
   596→    }, onEvent);
   597→
   598→    updatedContext.lastIntent = classification.intent;
   599→    updatedContext.lastPersona = classification.persona;
   600→
   601→    const agentMessage: AgentMessage = {
   602→      id: `agent-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`,
   603→      role: 'assistant',
   604→      content: responseContent,
   605→      timestamp: new Date(),
   606→      persona: classification.persona,
   607→      thinking,
   608→      actions: [{ type: classification.intent, label: 'Direct Response', status: 'completed', message: 'Fast-path response' }],
   609→    };
   610→
   611→    const suggestedActions = generateSuggestedActions(classification.intent, undefined, updatedContext);
   612→
   613→    sendCommMsg(pipelineState, 'atlas', 'user', 'response',
   614→      `Fast path: ${classification.intent} handled directly.`, undefined, onEvent);
   615→
   616→    console.log(`[Orchestrator] Fast-path "${userMessage.slice(0, 50)}" → intent=${classification.intent}, took=${Date.now() - startTime}ms`);
   617→
   618→    return { message: agentMessage, updatedContext, suggestedActions, pipelineState };
   619→  }
   620→
   621→  // ═══════════════════════════════════════════════════
   622→  // PHASE 2: EXECUTE — Run the 8-agent pipeline
   623→  // ═══════════════════════════════════════════════════
   624→  pipelineState.phase = 'executing';
   625→  emit(onEvent, { type: 'pipeline_progress', data: { phase: 'executing', overallProgress: 15 } });
   626→
   627→  let actions: AgentAction[] = [];
   628→  let prospectData: ProspectResult | undefined;
   629→  let icpData: ICPResult | undefined;
   630→  let outreachData: OutreachResult | undefined;
   631→  let marketData: MarketResult | undefined;
   632→  let scoreData: ScoreResult | undefined;
   633→  let responseContent = '';
   634→
   635→  // Initialize partial data from checkpoint if resuming
   636→  if (resumeFrom) {
   637→    if (resumeFrom.partialProspectData) prospectData = resumeFrom.partialProspectData as unknown as ProspectResult;
   638→    if (resumeFrom.partialIcpData) icpData = resumeFrom.partialIcpData as unknown as ICPResult;
   639→    if (resumeFrom.partialScoreData) scoreData = resumeFrom.partialScoreData as unknown as ScoreResult;
   640→    if (resumeFrom.partialOutreachData) outreachData = resumeFrom.partialOutreachData as unknown as OutreachResult;
   641→    if (resumeFrom.partialMarketData) marketData = resumeFrom.partialMarketData as unknown as MarketResult;
   642→    // Also restore context from partial data
   643→    if (prospectData) {
   644→      updatedContext.recentProspects = [...updatedContext.recentProspects, prospectData];
   645→    }
   646→  }
   647→
   648→  // Create a progress callback that bridges to the orchestrator events
   649→  const bridgeProgress: ProgressCallback = (event: string, data: unknown) => {
   650→    if (event === 'step_start') {
   651→      emit(onEvent, { type: 'step_start', data: data as { stepIndex: number; label: string; agent: AgentPersona; message: string } });
   652→    } else if (event === 'step_progress') {
   653→      emit(onEvent, { type: 'step_progress', data: data as { stepIndex: number; message: string; partialData?: Record<string, unknown> } });
   654→    } else if (event === 'step_complete') {
   655→      emit(onEvent, { type: 'step_complete', data: data as { stepIndex: number; status: 'completed' | 'failed'; message: string; partialData?: Record<string, unknown> } });
   656→    } else if (event === 'insight') {
   657→      emit(onEvent, { type: 'insight', data: data as { insight: InsightItem } });
   658→    }
   659→  };
   660→
   661→  // Determine which pipeline phases to execute based on intent
   662→  const relevantPhases = PIPELINE_PHASES.filter(phase => {
   663→    if (!phase.intentRequired) return true; // Always run (atlas, echo)
   664→    return phase.intentRequired.includes(classification.intent);
   665→  });
   666→
   667→  let stepIdx = 0;
   668→  const totalPhases = relevantPhases.length;
   669→
   670→  // Execute each relevant phase
   671→  for (const phase of relevantPhases) {
   672→    const agentKey = phase.agent; // 8-agent display name (atlas, scout, forge, etc.)
   673→
   674→    // ═══ SKIP COMPLETED AGENTS ON RESUME ═══
   675→    // If resuming from a checkpoint, skip agents that already completed
   676→    if (resumeFrom && resumeFrom.completedAgents.includes(phase.agent)) {
   677→      updateAgentState(pipelineState, agentKey, {
   678→        status: 'completed',
   679→        currentStep: `${phase.action} (completed before)`,
   680→        progress: 100,
   681→        completedAt: Date.now(),
   682→      }, onEvent);
   683→      stepIdx++;
   684→      continue;
   685→    }
   686→
   687→    // ═══ REDUCED COOLDOWN ═══
   688→    // Cooldown between agent phases to respect rate limit (concurrency=1).
   689→    // Reduced from 2-3.5s to 0.5-1.5s for faster pipeline execution.
   690→    if (stepIdx > 0) {
   691→      const isSearchPhase = phase.agent === 'scout';
   692→      const cooldownMs = isSearchPhase
   693→        ? 500 + Math.random() * 500    // 0.5-1s for search phases
   694→        : 1000 + Math.random() * 500;  // 1-1.5s for LLM phases
   695→
   696→      updateAgentState(pipelineState, agentKey, {
   697→        status: 'waiting',
   698→        currentStep: `Cooldown (${Math.round(cooldownMs / 1000)}s)`,
   699→        progress: 0,
   700→        startedAt: Date.now(),
   701→      }, onEvent);
   702→
   703→      sendCommMsg(pipelineState, 'atlas', agentKey, 'status',
   704→        `Cooldown: ${Math.round(cooldownMs / 1000)}s before starting`,
   705→        { cooldownMs, reason: 'rate_limit_buffer' },
   706→        onEvent);
   707→
   708→      emit(onEvent, { type: 'cooldown', data: { agent: agentKey, cooldownMs, reason: 'rate_limit_buffer' } });
   709→      emit(onEvent, { type: 'pipeline_progress', data: { phase: 'executing', overallProgress: Math.round(15 + (stepIdx / totalPhases) * 70 - 2) } });
   710→
   711→      await new Promise(r => setTimeout(r, cooldownMs));
   712→    }
   713→
   714→    // Update agent state to working
   715→    updateAgentState(pipelineState, agentKey, {
   716→      status: 'working',
   717→      currentStep: phase.action,
   718→      progress: 0,
   719→      startedAt: Date.now(),
   720→    }, onEvent);
   721→
   722→    // Atlas tells the agent what to do
   723→    if (phase.agent !== 'atlas') {
   724→      sendCommMsg(pipelineState, 'atlas', agentKey, 'request',
   725→        `[${agentKey.toUpperCase()}] ${phase.action} for query: "${userMessage.slice(0, 80)}"`,
   726→        { intent: classification.intent, phase: agentKey },
   727→        onEvent);
   728→    }
   729→
   730→    // Progress calculation
   731→    const baseProgress = 15 + (stepIdx / totalPhases) * 70;
   732→    emit(onEvent, { type: 'pipeline_progress', data: { phase: 'executing', overallProgress: Math.round(baseProgress) } });
   733→
   734→    // Execute the actual work based on the agent
   735→    try {
   736→      switch (phase.agent) {
   737→        case 'atlas':
   738→          // Already done (intent classification)
   739→          sendCommMsg(pipelineState, 'atlas', 'all', 'broadcast',
   740→            `Pipeline plan: ${relevantPhases.map(p => p.agent).join(' → ')}`, undefined, onEvent);
   741→          break;
   742→
   743→        case 'scout':
   744→          // Scout: Company/Person/URL research
   745→          if (['research_company', 'research_url', 'refine_search'].includes(classification.intent)) {
   746→            const companyName = classification.extractedEntities.companyName || userMessage.trim();
   747→            sendCommMsg(pipelineState, 'atlas', 'scout', 'handoff',
   748→              `Research target: "${companyName}"`, { target: companyName }, onEvent);
   749→
   750→            // Emit an immediate data_update with pre-populated data so the UI
   751→            // can render user-supplied fields BEFORE the slow agent pipeline
   752→            // finishes. This fixes the "all fields say Discovering..." problem.
   753→            if (classification.prepopulatedProspect && Object.keys(classification.prepopulatedProspect).length > 0) {
   754→              emit(onEvent, { type: 'data_update', data: { prospect: classification.prepopulatedProspect } });
   755→            }
   756→
   757→            const result = await executeCompanyResearch(
   758→              companyName,
   759→              bridgeProgress,
   760→              classification.prepopulatedProspect,
   761→            );
   762→            actions = [...actions, ...result.steps];
   763→
   764→            if (result.prospect) {
   765→              prospectData = result.prospect;
   766→              updatedContext.recentProspects = [...updatedContext.recentProspects.slice(-4), result.prospect];
   767→
   768→              // Final data_update with the complete prospect data
   769→              emit(onEvent, { type: 'data_update', data: { prospect: prospectData as unknown as Record<string, unknown> } });
   770→
   771→              sendCommMsg(pipelineState, 'scout', 'atlas', 'response',
   772→                `Found data for "${result.prospect.companyName}" — ${result.prospect.dataCompleteness}% complete`,
   773→                { completeness: result.prospect.dataCompleteness, companyName: result.prospect.companyName },
   774→                onEvent);
   775→            } else {
   776→              sendCommMsg(pipelineState, 'scout', 'atlas', 'response',
   777→                'Limited data found. Continuing with partial results.', undefined, onEvent);
   778→            }
   779→          } else if (classification.intent === 'research_person') {
   780→            const personName = classification.extractedEntities.personName || userMessage.trim();
   781→            sendCommMsg(pipelineState, 'atlas', 'scout', 'handoff',
   782→              `Research person: "${personName}"`, { target: personName }, onEvent);
   783→
   784→            // Emit an immediate data_update with pre-populated data so the UI
   785→            // can render user-supplied fields BEFORE the slow agent pipeline
   786→            // finishes.
   787→            if (classification.prepopulatedProspect && Object.keys(classification.prepopulatedProspect).length > 0) {
   788→              emit(onEvent, { type: 'data_update', data: { prospect: classification.prepopulatedProspect } });
   789→            }
   790→
   791→            const result = await executePersonResearch(
   792→              personName,
   793→              bridgeProgress,
   794→              classification.prepopulatedProspect,
   795→            );
   796→            actions = [...actions, ...result.steps];
   797→            if (result.prospect) {
   798→              prospectData = result.prospect;
   799→              updatedContext.recentProspects = [...updatedContext.recentProspects.slice(-4), result.prospect];
   800→
   801→              // Final data_update with the complete prospect data
   802→              emit(onEvent, { type: 'data_update', data: { prospect: prospectData as unknown as Record<string, unknown> } });
   803→
   804→              sendCommMsg(pipelineState, 'scout', 'atlas', 'response',
   805→                `Found data for "${result.prospect.personName}" — ${result.prospect.dataCompleteness}% complete`,
   806→                { completeness: result.prospect.dataCompleteness, personName: result.prospect.personName },
   807→                onEvent);
   808→            }
   809→          }
   810→          break;
   811→
   812→        case 'forge':
   813→          // Forge: Data enrichment — deep crawl and gap fill
   814→          if (prospectData && prospectData.dataCompleteness < 80) {
   815→            sendCommMsg(pipelineState, 'scout', 'forge', 'handoff',
   816→              `Enriching data (currently ${prospectData.dataCompleteness}% complete)`,
   817→              { completeness: prospectData.dataCompleteness }, onEvent);
   818→
   819→            // Try additional enrichment searches
   820→            const enrichActions: AgentAction = {
   821→              type: 'research_company', label: 'Deep Enrichment', status: 'running',
   822→              message: `Enriching ${prospectData.companyName || 'prospect'} data...`,
   823→            };
   824→            actions.push(enrichActions);
   825→            const enrichIdx = actions.length - 1;
   826→
   827→            try {
   828→              // Try LinkedIn search for enrichment
   829→              if (prospectData.companyName) {
   830→                const liResult = await linkedInSearchCompanies(prospectData.companyName, 3);
   831→                if (liResult.success && liResult.data.length > 0) {
   832→                  const topResult = liResult.data[0];
   833→                  if (!prospectData.linkedinUrl && topResult.url) prospectData.linkedinUrl = topResult.url;
   834→                  if (!prospectData.description && topResult.headline) prospectData.description = topResult.headline;
   835→                }
   836→              }
   837→
   838→              // Try web search for missing data points
   839→              if (prospectData.companyName) {
   840→                const gapQueries: string[] = [];
   841→                if (!prospectData.ceoName) gapQueries.push(`"${prospectData.companyName}" CEO founder executive`);
   842→                if (!prospectData.revenueEstimate) gapQueries.push(`"${prospectData.companyName}" revenue funding valuation`);
   843→                if (!prospectData.employeeCount) gapQueries.push(`"${prospectData.companyName}" employees team size`);
   844→
   845→                if (gapQueries.length > 0) {
   846→                  const gapSearch = await exaSearch(gapQueries[0], 3);
   847→                  if (gapSearch.success && gapSearch.data.length > 0) {
   848→                    const allText = gapSearch.data.map(r => `${r.title} ${r.snippet}`).join(' ');
   849→
   850→                    // Simple regex extraction for common missing fields
   851→                    if (!prospectData.ceoName) {
   852→                      const ceoMatch = allText.match(/(?:CEO|Chief Executive|Founder)[:\s]+([A-Z][a-z]+ [A-Z][a-z]+)/);
   853→                      if (ceoMatch) prospectData.ceoName = ceoMatch[1];
   854→                    }
   855→                    if (!prospectData.revenueEstimate) {
   856→                      const revMatch = allText.match(/\$[\d.]+[BbMmKk]/);
   857→                      if (revMatch) prospectData.revenueEstimate = revMatch[0];
   858→                    }
   859→                    if (!prospectData.employeeCount) {
   860→                      const empMatch = allText.match(/(\d[\d,+]*)\s*(?:employees|team members|staff|people)/i);
   861→                      if (empMatch) prospectData.employeeCount = empMatch[1];
   862→                    }
   863→                  }
   864→                }
   865→              }
   866→
   867→              // Recalculate completeness
   868→              prospectData.dataCompleteness = calculateQuickCompleteness(prospectData);
   869→
   870→              actions[enrichIdx] = { ...enrichActions, status: 'completed', message: `Enrichment complete — ${prospectData.dataCompleteness}% data` };
   871→              sendCommMsg(pipelineState, 'forge', 'atlas', 'response',
   872→                `Enrichment brought data to ${prospectData.dataCompleteness}%`, undefined, onEvent);
   873→            } catch (e) {
   874→              actions[enrichIdx] = { ...enrichActions, status: 'completed', message: 'Enrichment partially completed' };
   875→            }
   876→
   877→            // ── Phase 1 Data Source Channels ──────────────────────────────
   878→            // Run OpenCorporates, yfinance, SEC EDGAR, News Worker, Geocoder,
   879→            // and PublicWWW in parallel to enrich the prospect with verified
   880→            // registry data, financial KPIs, news intent signals, and geo coords.
   881→            // PublicWWW is skipped here to respect its 60s rate limit — it's
   882→            // only called explicitly from the Scout phase for tech-stack discovery.
   883→            if (prospectData && (prospectData.companyName || prospectData.personName)) {
   884→              try {
   885→                const channelActions: AgentAction = {
   886→                  type: 'research_company',
   887→                  label: 'Data Source Channels',
   888→                  status: 'running',
   889→                  message: `Querying OpenCorporates, yfinance, SEC EDGAR, News, Geocoder...`,
   890→                };
   891→                actions.push(channelActions);
   892→                const chIdx = actions.length - 1;
   893→
   894→                const channelResult = await executeChannelEnrichment(prospectData, {
   895→                  onProgress: (label, message, status) => {
   896→                    if (status === 'running') {
   897→                      actions[chIdx] = { ...channelActions, status: 'running', message };
   898→                    } else if (status === 'completed') {
   899→                      actions[chIdx] = { ...channelActions, status: 'completed', message };
   900→                    }
   901→                  },
   902→                  skipPublicWww: true,  // 60s rate limit — call separately in Scout phase
   903→                });
   904→
   905→                // Replace prospectData with the enriched version
   906→                prospectData = channelResult.prospect;
   907→
   908→                const ocOk = channelResult.channelResults.openCorporates?.success ? '✓' : '✗';
   909→                const yfOk = channelResult.channelResults.yfinance?.success ? '✓' : '✗';
   910→                const edOk = channelResult.channelResults.edgar?.success ? '✓' : '✗';
   911→                const nwOk = channelResult.channelResults.news?.success ? '✓' : '✗';
   912→                const geOk = channelResult.channelResults.geocoder?.success ? '✓' : '✗';
   913→
   914→                actions[chIdx] = {
   915→                  ...channelActions,
   916→                  status: 'completed',
   917→                  message: `Channels: OC:${ocOk} YF:${yfOk} EDGAR:${edOk} News:${nwOk} Geo:${geOk} — ` +
   918→                           `${channelResult.personLeads.length} person leads, ` +
   919→                           `${Object.keys(channelResult.customKpis).length} KPIs`,
   920→                };
   921→
   922→                sendCommMsg(pipelineState, 'forge', 'atlas', 'response',
   923→                  `Channel enrichment: ${Object.keys(channelResult.customKpis).length} KPIs, ` +
   924→                  `${channelResult.personLeads.length} person leads`,
   925→                  { kpiCount: Object.keys(channelResult.customKpis).length,
   926→                    personLeadCount: channelResult.personLeads.length },
   927→                  onEvent);
   928→
   929→                // Recalculate completeness after channel enrichment
   930→                prospectData.dataCompleteness = calculateQuickCompleteness(prospectData);
   931→              } catch (e) {
   932→                // Non-fatal — the pipeline continues with whatever data we have
   933→                console.warn('[orchestrator] Channel enrichment failed:', e);
   934→              }
   935→            }
   936→          }
   937→          break;
   938→
   939→        case 'sage':
   940→          // Sage: Market/Competitive analysis
   941→          if (classification.intent === 'analyze_market') {
   942→            const result = await executeMarketAnalysis(userMessage);
   943→            actions = [...actions, ...result.steps];
   944→            if (result.market) {
   945→              marketData = result.market;
   946→              sendCommMsg(pipelineState, 'sage', 'atlas', 'response',
   947→                `Market analysis complete: ${result.market.keyFindings.length} findings, ${result.market.competitors.length} competitors`,
   948→                { findings: result.market.keyFindings.length, competitors: result.market.competitors.length },
   949→                onEvent);
   950→            }
   951→          } else if (classification.intent === 'analyze_competitors') {
   952→            const result = await executeCompetitiveAnalysis(userMessage);
   953→            actions = [...actions, ...result.steps];
   954→            if (result.market) {
   955→              marketData = result.market;
   956→              sendCommMsg(pipelineState, 'sage', 'atlas', 'response',
   957→                `Competitive analysis complete`,
   958→                undefined, onEvent);
   959→            }
   960→          }
   961→          break;
   962→
   963→        case 'judge':
   964→          // Judge: Lead scoring
   965→          {
   966→            const recentProspect = updatedContext.recentProspects[updatedContext.recentProspects.length - 1];
   967→            if (recentProspect) {
   968→              sendCommMsg(pipelineState, 'atlas', 'judge', 'request',
   969→                `Score ${recentProspect.companyName || recentProspect.personName} against ICP`,
   970→                { companyName: recentProspect.companyName }, onEvent);
   971→
   972→              // Refresh news intent signals right before scoring — this gives the
   973→              // Judge the freshest possible view of recent news/market activity.
   974→              try {
   975→                const targetName = recentProspect.companyName || recentProspect.personName || '';
   976→                if (targetName) {
   977→                  const newsRefresh = await refreshNewsIntent(targetName);
   978→                  if (newsRefresh.intent && newsRefresh.intent.success) {
   979→                    const kpiKeys = Object.keys(newsRefresh.kpis);
   980→                    if (kpiKeys.length > 0) {
   981→                      // Merge fresh news KPIs into prospect's customKpis
   982→                      const existingKpis = recentProspect.customKpis
   983→                        ? JSON.parse(recentProspect.customKpis)
   984→                        : {};
   985→                      const merged = { ...existingKpis, ...newsRefresh.kpis };
   986→                      recentProspect.customKpis = JSON.stringify(merged);
   987→
   988→                      sendCommMsg(pipelineState, 'judge', 'atlas', 'response',
   989→                        `Refreshed news intent: ${newsRefresh.intent.mentionCount30d} mentions, ` +
   990→                        `${kpiKeys.length} signals (sentiment: ${newsRefresh.intent.sentiment.overall})`,
   991→                        { mentions: newsRefresh.intent.mentionCount30d, sentiment: newsRefresh.intent.sentiment.overall },
   992→                        onEvent);
   993→                    }
   994→                  }
   995→                }
   996→              } catch (e) {
   997→                console.warn('[orchestrator] News refresh in Judge phase failed:', e);
   998→              }
   999→
  1000→              const result = await executeLeadScoring(recentProspect, updatedContext.activeICP);
  1001→              actions = [...actions, ...result.steps];
  1002→              if (result.score) {
  1003→                scoreData = result.score;
  1004→                sendCommMsg(pipelineState, 'judge', 'atlas', 'response',
  1005→                  `Lead score: ${result.score.overallScore}/100 (${result.score.tier})`,
  1006→                  { score: result.score.overallScore, tier: result.score.tier },
  1007→                  onEvent);
  1008→              }
  1009→            } else if (classification.intent === 'score_lead') {
  1010→              responseContent = "I don't have a prospect to score yet. Please research a company or person first, then I can score them against your ICP.";
  1011→            }
  1012→          }
  1013→          break;
  1014→
  1015→        case 'bard':
  1016→          // Bard: Outreach composition
  1017→          {
  1018→            const recentProspect = updatedContext.recentProspects[updatedContext.recentProspects.length - 1];
  1019→            if (recentProspect) {
  1020→              const channel = userMessage.toLowerCase().includes('linkedin') ? 'linkedin' : 'email';
  1021→              sendCommMsg(pipelineState, 'atlas', 'bard', 'request',
  1022→                `Compose ${channel} outreach for ${recentProspect.companyName || recentProspect.personName}`,
  1023→                { channel, target: recentProspect.companyName }, onEvent);
  1024→
  1025→              const result = await executeOutreachComposition(recentProspect, channel);
  1026→              actions = [...actions, ...result.steps];
  1027→              if (result.outreach) {
  1028→                outreachData = result.outreach;
  1029→                sendCommMsg(pipelineState, 'bard', 'atlas', 'response',
  1030→                  `${channel} outreach composed with ${result.outreach.personalizationHooks?.length || 0} personalization hooks`,
  1031→                  { channel, hooks: result.outreach.personalizationHooks?.length },
  1032→                  onEvent);
  1033→              }
  1034→            } else if (classification.intent === 'compose_outreach') {
  1035→              const target = userMessage.replace(/^(?:write|compose|draft|send|email|reach out|craft)\s*(?:an?\s*)?(?:email|message|outreach|connection)?\s*(?:to\s*)?/i, '').trim();
  1036→              if (target.length > 2) {
  1037→                responseContent = `I need to research "${target}" first to write a personalized message. Would you like me to research them?`;
  1038→              } else {
  1039→                responseContent = "I need a prospect to compose outreach for. Please research a company or person first.";
  1040→              }
  1041→            }
  1042→          }
  1043→          break;
  1044→
  1045→        case 'flow':
  1046→          // Flow: Pipeline management — saves session context and manages pipeline state
  1047→          if (classification.intent === 'build_icp') {
  1048→            const result = await executeICPBuilding(userMessage, updatedContext.activeICP);
  1049→            actions = [...actions, ...result.steps];
  1050→            if (result.icp) {
  1051→              icpData = result.icp;
  1052→              updatedContext.activeICP = result.icp;
  1053→              responseContent = result.response;
  1054→            }
  1055→          } else if (classification.intent === 'add_to_pipeline') {
  1056→            const recentProspect = updatedContext.recentProspects[updatedContext.recentProspects.length - 1];
  1057→            const name = recentProspect?.companyName || recentProspect?.personName || 'this prospect';
  1058→            responseContent = `Click the "Add to Leads" button below to add ${name} to your lead pipeline.`;
  1059→          } else {
  1060→            // Flow manages pipeline context for all other intents
  1061→            const prospectCount = updatedContext.recentProspects.length;
  1062→            const hasICP = !!updatedContext.activeICP;
  1063→            const hasScoredLeads = updatedContext.recentProspects.some(p => (p as unknown as Record<string, unknown>).leadScore !== undefined);
  1064→            let flowMsg = `Pipeline context: ${prospectCount} prospect(s) discovered`;
  1065→            if (hasICP) flowMsg += `, ICP active`;
  1066→            if (hasScoredLeads) flowMsg += `, leads scored`;
  1067→            flowMsg += '. ';
  1068→            if (prospectCount > 0 && !hasICP) {
  1069→              flowMsg += 'Suggest building an ICP to improve qualification.';
  1070→            } else if (hasScoredLeads) {
  1071→              flowMsg += 'Recommend composing outreach for scored leads.';
  1072→            } else if (prospectCount > 0) {
  1073→              flowMsg += 'Recommend scoring prospects against ICP.';
  1074→            }
  1075→            sendCommMsg(pipelineState, 'flow', 'atlas', 'response', flowMsg, undefined, onEvent);
  1076→          }
  1077→          break;
  1078→
  1079→        case 'echo':
  1080→          // Echo: Insights and reporting — compiles final insights from all agents
  1081→          {

  1083→            if (prospectData) {
  1084→              echoInsights.push(`Prospect research completed for ${prospectData.companyName || prospectData.personName || 'target'}`);
  1085→              if (prospectData.employeeCount || prospectData.revenueEstimate) {
  1086→                echoInsights.push(`Firmographics: ${[prospectData.employeeCount ? `${prospectData.employeeCount} employees` : '', prospectData.revenueEstimate ? `~${prospectData.revenueEstimate} revenue` : ''].filter(Boolean).join(', ')}`);
  1087→              }
  1088→            }
  1089→            if (marketData) {
  1090→              echoInsights.push(`Market analysis available with ${marketData.competitors?.length || 0} competitors identified`);
  1091→            }
  1092→            if (scoreData) {
  1093→              echoInsights.push(`Lead scored: ${scoreData.overallScore}/100 (${scoreData.tier || 'unrated'} tier)`);
  1094→            }
  1095→            if (outreachData) {
  1096→              echoInsights.push(`Outreach message composed`);
  1097→            }
  1098→            if (icpData) {
  1099→              echoInsights.push(`ICP profile: ${icpData.name || 'Active ICP'}`);
  1100→            } else if (updatedContext.activeICP) {
  1101→              icpData = updatedContext.activeICP;
  1102→              echoInsights.push(`Existing ICP included: ${updatedContext.activeICP.name || 'Active ICP'}`);
  1103→            }
  1104→
  1105→            if (echoInsights.length > 0) {
  1106→              const echoReport = echoInsights.join('. ') + '.';

  1110→            }
  1111→          }
  1112→          break;
  1113→      }
  1114→    } catch (phaseError) {
  1115→      const msg = phaseError instanceof Error ? phaseError.message : 'Unknown error';
  1116→      console.warn(`[Orchestrator] Phase ${phase.agent} failed: ${msg}`);
  1117→
  1118→      if (!phase.optional) {
  1119→        // Non-optional phase failed — still continue with other phases
  1120→      }
  1121→      updateAgentState(pipelineState, agentKey, {
  1122→        status: 'failed',
  1123→        currentStep: phase.action,
  1124→        completedAt: Date.now(),
  1125→      }, onEvent);
  1126→
  1127→      sendCommMsg(pipelineState, agentKey, 'atlas', 'response',
  1128→        `Phase failed: ${msg.slice(0, 100)}`, { error: msg.slice(0, 200) }, onEvent);
  1129→    }
  1130→
  1131→    // Update agent state to completed
  1132→    updateAgentState(pipelineState, agentKey, {
  1133→      status: 'completed',
  1134→      currentStep: phase.action,
  1135→      progress: 100,
  1136→      completedAt: Date.now(),
  1137→    }, onEvent);
  1138→
  1139→    stepIdx++;
  1140→  }
  1141→
  1142→  // ═══════════════════════════════════════════════════
  1143→  // PHASE 3: SYNTHESIZE — Generate conversational response
  1144→  // ═══════════════════════════════════════════════════
  1145→  pipelineState.phase = 'synthesizing';
  1146→  emit(onEvent, { type: 'pipeline_progress', data: { phase: 'synthesizing', overallProgress: 85 } });
  1147→
  1148→  // Reduced cooldown before synthesis (0.5-1s instead of 1.5-2.5s)
  1149→  const synthCooldownMs = 500 + Math.random() * 500;
  1150→  updateAgentState(pipelineState, 'atlas', {
  1151→    status: 'waiting',
  1152→    currentStep: `Cooldown before synthesis (${Math.round(synthCooldownMs / 1000)}s)`,
  1153→    progress: 0,
  1154→    startedAt: Date.now(),
  1155→  }, onEvent);
  1156→
  1157→  emit(onEvent, { type: 'cooldown', data: { agent: 'atlas', cooldownMs: synthCooldownMs, reason: 'rate_limit_buffer' } });
  1158→
  1159→  await new Promise(r => setTimeout(r, synthCooldownMs));
  1160→
  1161→  updateAgentState(pipelineState, 'atlas', {
  1162→    status: 'working',
  1163→    currentStep: 'Synthesizing response',
  1164→    progress: 50,
  1165→    startedAt: Date.now(),
  1166→  }, onEvent);
  1167→
  1168→  sendCommMsg(pipelineState, 'atlas', 'echo', 'request',
  1169→    'Synthesize all agent outputs into a coherent response', undefined, onEvent);
  1170→
  1171→  // Generate conversational response with robust fallback chain
  1172→  if (!responseContent) {
  1173→    if (prospectData) {
  1174→      const actionSummary = buildResearchSummary(prospectData);
  1175→      try {
  1176→        responseContent = await generateConversationResponse(
  1177→          classification.persona, classification.intent, userMessage, actionSummary, updatedContext,
  1178→        );
  1179→      } catch { /* LLM unavailable — will use fallback */ }
  1180→      if (!responseContent) {
  1181→        responseContent = buildFallbackResponse(prospectData, classification.intent);
  1182→      }
  1183→    } else if (marketData) {
  1184→      const actionSummary = JSON.stringify({
  1185→        summary: marketData.summary,
  1186→        findings: marketData.keyFindings,
  1187→        competitors: marketData.competitors.map(c => c.name),
  1188→        trends: marketData.trends,
  1189→      });
  1190→      try {
  1191→        responseContent = await generateConversationResponse(
  1192→          classification.persona, classification.intent, userMessage, actionSummary, updatedContext,
  1193→        );
  1194→      } catch { /* LLM unavailable — will use fallback */ }
  1195→      if (!responseContent) {
  1196→        responseContent = buildRichFallbackResponse({ intent: classification.intent, market: marketData, userMessage });
  1197→      }
  1198→    } else if (scoreData) {
  1199→      const actionSummary = JSON.stringify(scoreData);
  1200→      try {
  1201→        responseContent = await generateConversationResponse(
  1202→          classification.persona, classification.intent, userMessage, actionSummary, updatedContext,
  1203→        );
  1204→      } catch { /* LLM unavailable — will use fallback */ }
  1205→      if (!responseContent) {
  1206→        responseContent = buildRichFallbackResponse({ intent: classification.intent, score: scoreData, userMessage });
  1207→      }
  1208→    } else if (outreachData) {
  1209→      const actionSummary = JSON.stringify(outreachData);
  1210→      try {
  1211→        responseContent = await generateConversationResponse(
  1212→          classification.persona, classification.intent, userMessage, actionSummary, updatedContext,
  1213→        );
  1214→      } catch { /* LLM unavailable — will use fallback */ }

  1218→    } else if (icpData) {
  1219→      responseContent = buildRichFallbackResponse({ intent: classification.intent, icp: icpData, userMessage });

  1221→    // Note: 'clarify' intent is handled in the fast path above, before the full pipeline
  1222→  }
  1223→
  1224→  // Fallback using structured data (no LLM needed)
  1225→  if (!responseContent && (prospectData || icpData || marketData || scoreData || outreachData)) {
  1226→    responseContent = generateStructuredFallback({
  1227→      persona: classification.persona,
  1228→      intent: classification.intent,
  1229→      userMessage,
  1230→      actionSummary: prospectData ? buildResearchSummary(prospectData) :
  1231→                     icpData ? JSON.stringify(icpData) :
  1232→                     marketData ? JSON.stringify(marketData) :
  1233→                     scoreData ? JSON.stringify(scoreData) :
  1234→                     outreachData ? JSON.stringify(outreachData) : '{}',
  1235→      context: buildContextHint(updatedContext),
  1236→    });
  1237→  }
  1238→
  1239→  // Final fallback
  1240→  if (!responseContent) {

  1242→  }
  1243→
  1244→  // ═══════════════════════════════════════════════════
  1245→  // PHASE 4: COMPLETE — Generate insights & finalize
  1246→  // ═══════════════════════════════════════════════════
  1247→  pipelineState.phase = 'complete';
  1248→  emit(onEvent, { type: 'pipeline_progress', data: { phase: 'complete', overallProgress: 100 } });
  1249→
  1250→  // Generate insights
  1251→  const insights = generateInsights(classification.intent, prospectData, icpData, scoreData, marketData, outreachData, updatedContext);
  1252→  if (onEvent && insights.length > 0) {
  1253→    for (const insight of insights) {
  1254→      emit(onEvent, { type: 'insight', data: { insight } });
  1255→    }
  1256→  }
  1257→
  1258→  // Generate navigation suggestions
  1259→  const navigation = generateNavigationSuggestions(classification.intent, prospectData, icpData, scoreData, outreachData, marketData);
  1260→
  1261→  // Update context
  1262→  updatedContext.lastIntent = classification.intent;
  1263→  updatedContext.lastPersona = classification.persona;
  1264→
  1265→  // Build the agent message
  1266→  const agentMessage: AgentMessage = {
  1267→    id: `agent-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`,
  1268→    role: 'assistant',
  1269→    content: responseContent,
  1270→    timestamp: new Date(),
  1271→    persona: classification.persona,
  1272→    thinking,
  1273→    actions,
  1274→    prospectData,
  1275→    icpData,
  1276→    outreachData,
  1277→    marketData,
  1278→    scoreData,
  1279→    insights: insights.length > 0 ? insights : undefined,
  1280→    navigation: navigation.length > 0 ? navigation : undefined,
  1281→  };
  1282→
  1283→  // Generate suggested actions
  1284→  const suggestedActions = generateSuggestedActions(classification.intent, prospectData, updatedContext);
  1285→
  1286→  // Final comm message
  1287→  sendCommMsg(pipelineState, 'atlas', 'user', 'response',
  1288→    `Pipeline complete. ${actions.length} steps executed, ${actions.filter(a => a.status === 'completed').length} succeeded.`,
  1289→    { steps: actions.length, completed: actions.filter(a => a.status === 'completed').length },
  1290→    onEvent);
  1291→
  1292→  console.log(`[Orchestrator] Processed "${userMessage.slice(0, 50)}" → intent=${classification.intent}, took=${Date.now() - startTime}ms`);
  1293→
  1294→  return {
  1295→    message: agentMessage,
  1296→    updatedContext,
  1297→    suggestedActions,
  1298→    pipelineState,
  1299→  };

  1306→function calculateQuickCompleteness(prospect: ProspectResult): number {
  1307→  const fields: Array<{ key: keyof ProspectResult; weight: number }> = [
  1308→    { key: 'companyName', weight: 10 },
  1309→    { key: 'website', weight: 8 },
  1310→    { key: 'industry', weight: 8 },
  1311→    { key: 'description', weight: 6 },
  1312→    { key: 'ceoName', weight: 7 },
  1313→    { key: 'employeeCount', weight: 6 },
  1314→    { key: 'revenueEstimate', weight: 6 },
  1315→    { key: 'generalEmail', weight: 6 },
  1316→    { key: 'phoneMain', weight: 5 },
  1317→    { key: 'linkedinUrl', weight: 5 },
  1318→    { key: 'city', weight: 4 },
  1319→    { key: 'country', weight: 4 },
  1320→    { key: 'foundingYear', weight: 3 },
  1321→    { key: 'twitterHandle', weight: 3 },
  1322→    { key: 'fundingInfo', weight: 4 },
  1323→    { key: 'techStack', weight: 5 },
  1324→    { key: 'recentNews', weight: 5 },
  1325→    { key: 'productsServices', weight: 5 },
  1326→  ];
  1327→
  1328→  let total = 0;
  1329→  let maxTotal = 0;
  1330→  for (const { key, weight } of fields) {
  1331→    maxTotal += weight;
  1332→    const val = prospect[key];
  1333→    if (val !== null && val !== undefined && val !== '' && !(Array.isArray(val) && val.length === 0)) {
  1334→      total += weight;
  1335→    }
  1336→  }
  1337→  return Math.round((total / maxTotal) * 100);
  1338→}

  1358→function buildFallbackResponse(prospect: ProspectResult, intent: UserIntent): string {
  1359→  const parts: string[] = [];

  1361→  if (prospect.description) parts.push(prospect.description);
  1362→  if (prospect.industry) parts.push(`**Industry:** ${prospect.industry}`);
  1363→  if (prospect.employeeCount) parts.push(`**Employees:** ${prospect.employeeCount}`);
  1364→  if (prospect.revenueEstimate) parts.push(`**Revenue:** ${prospect.revenueEstimate}`);
  1365→  if (prospect.ceoName) parts.push(`**CEO:** ${prospect.ceoName}`);
  1366→  if (prospect.city || prospect.country) parts.push(`**Location:** ${[prospect.city, prospect.country].filter(Boolean).join(', ')}`);
  1367→  if (prospect.website) parts.push(`**Website:** ${prospect.website}`);
  1368→  if (prospect.generalEmail) parts.push(`**Email:** ${prospect.generalEmail}`);
  1369→  if (prospect.linkedinUrl) parts.push(`**LinkedIn:** available`);
  1370→  if (prospect.techStack?.length) parts.push(`**Tech Stack:** ${prospect.techStack.join(', ')}`);
  1371→  if (prospect.recentNews?.length) parts.push(`**Recent News:**\n${prospect.recentNews.slice(0, 3).map(n => `- ${n}`).join('\n')}`);
  1372→  if (prospect.detectedDomain && prospect.detectedDomain !== 'general') {
  1373→    parts.push(`\n**Domain:** ${prospect.domainLabel || prospect.detectedDomain} — 4-Phase Pipeline Active`);
  1374→  }
  1375→  parts.push(`\n*Data completeness: ${prospect.dataCompleteness}%*`);
  1376→  if (prospect.dataCompleteness >= 60) {
  1377→    parts.push('Would you like me to score this lead, compose outreach, or find similar companies?');
  1378→  }
  1379→  return parts.join('\n\n');
  1380→}

  1473→function generateSuggestedActions(
  1474→  intent: UserIntent,
  1475→  prospect?: ProspectResult,
  1476→  context?: ConversationContext,
  1477→): SuggestedAction[] {
  1478→  const actions: SuggestedAction[] = [];
  1479→  const hasICP = !!context?.activeICP;
  1480→  const prospectName = prospect?.companyName || prospect?.personName;
  1481→
  1482→  switch (intent) {
  1483→    case 'research_company':
  1484→    case 'research_person':
  1485→    case 'research_url':
  1486→      if (prospect) {
  1487→        if (hasICP) {
  1488→          actions.push({ label: 'Score Against ICP', prompt: `Score ${prospectName} against my ICP`, icon: 'Star' });
  1489→        } else {
  1490→          actions.push({ label: 'Build an ICP', prompt: 'Help me build an Ideal Customer Profile', icon: 'Target' });
  1491→        }
  1492→        actions.push({ label: 'Compose Outreach', prompt: `Write an email to ${prospectName}`, icon: 'Mail' });
  1493→        actions.push({ label: 'Add to Leads', prompt: 'Add this prospect to my leads', icon: 'Plus' });
  1494→        if (prospect.industry) {
  1495→          actions.push({ label: 'Find Similar', prompt: `Find similar companies in ${prospect.industry}`, icon: 'Search' });
  1496→        }
  1497→      }
  1498→      break;
  1499→
  1500→    case 'analyze_market':
  1501→    case 'analyze_competitors':
  1502→      actions.push({ label: 'Research Top Company', prompt: 'Research the top company from this analysis', icon: 'Building2' });
  1503→      if (!hasICP) actions.push({ label: 'Build an ICP', prompt: 'Build an ICP for this market', icon: 'Target' });
  1504→      break;
  1505→
  1506→    case 'score_lead':
  1507→      if (prospect) {
  1508→        actions.push({ label: 'Compose Outreach', prompt: `Write an email to ${prospectName}`, icon: 'Mail' });
  1509→        actions.push({ label: 'Add to Leads', prompt: 'Add this prospect to my leads', icon: 'Plus' });
  1510→      }
  1511→      break;
  1512→
  1513→    case 'compose_outreach':
  1514→      if (prospect) {
  1515→        actions.push({ label: 'Add to Leads', prompt: 'Add this prospect to my leads', icon: 'Plus' });
  1516→        actions.push({ label: 'Find Similar', prompt: 'Find similar companies', icon: 'Search' });
  1517→      }
  1518→      break;
  1519→
  1520→    default:
  1521→      actions.push({ label: 'Research a Company', prompt: 'I want to research a company', icon: 'Building2' });
  1522→      if (!hasICP) actions.push({ label: 'Build an ICP', prompt: 'Help me build an Ideal Customer Profile', icon: 'Target' });
  1523→      break;
  1524→  }
  1525→
  1526→  return actions.slice(0, 4);
  1527→}
