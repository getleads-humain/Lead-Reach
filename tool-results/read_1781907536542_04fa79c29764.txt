   336→/**
   337→ * Get the intent classification prompt.
   338→ * This is sent to the LLM to classify what the user wants to do.
   339→ * Enhanced with multi-intent detection and contextual awareness.
   340→ */
   341→export function getIntentClassificationPrompt(
   342→  userMessage: string,
   343→  context: ConversationContext | undefined,
   344→): string {
   345→  const contextStr = context ? `
   346→CONVERSATION CONTEXT:
   347→${context.lastIntent ? `- Last intent: ${context.lastIntent}` : ''}
   348→${context.lastPersona ? `- Last persona: ${context.lastPersona}` : ''}
   349→${context.activeICP ? `- Active ICP: ${context.activeICP.name}` : ''}
   350→${context.recentProspects.length > 0 ? `- Recent prospects discussed: ${context.recentProspects.map(p => p.companyName || p.personName).filter(Boolean).join(', ')}` : ''}
   351→${context.userPreferences.industries?.length ? `- User's industries of interest: ${context.userPreferences.industries.join(', ')}` : ''}
   352→${context.userPreferences.locations?.length ? `- User's locations of interest: ${context.userPreferences.locations.join(', ')}` : ''}
   353→${context.userPreferences.companySizes?.length ? `- User's preferred company sizes: ${context.userPreferences.companySizes.join(', ')}` : ''}
   354→` : '';
   355→
   356→  return `You are an intent classifier for a B2B lead generation AI agent with domain-specific intelligence. Classify the user's message into exactly ONE primary intent. Also detect the domain context and if the message implies a secondary intent that should be executed after the primary one.
   357→
   358→AVAILABLE INTENTS:
   359→- research_company: User wants to research, find information about, or discover a specific company
   360→- research_person: User wants to find information about a specific person (professional context)
   361→- research_url: User provided a URL and wants analysis of that website/page
   362→- analyze_market: User wants market/industry analysis, trends, or landscape overview
   363→- analyze_competitors: User wants competitive analysis or comparison between companies
   364→- build_icp: User wants to define, create, or refine an Ideal Customer Profile
   365→- score_lead: User wants to evaluate/qualify a lead or check if a company/person fits their ICP
   366→- compose_outreach: User wants to write an outreach message (email, LinkedIn, etc.)
   367→- refine_search: User wants to find more/similar results based on previous research
   368→- add_to_pipeline: User wants to add a discovered prospect to their leads database
   369→- clarify: The query is too vague or ambiguous to determine intent — needs clarification
   370→- converse: General conversation, follow-up questions, or non-actionable chat
   371→
   372→AVAILABLE DOMAINS (detect which domain the query falls into):
   373→- venture_capital: VC firms, funds, LPs, dry powder, fund performance, SEC Form ADV
   374→- private_equity: PE firms, buyout funds, LBO, EBITDA multiples, operating partners
   375→- hedge_funds: Hedge fund managers, AUM, Sharpe ratio, prime brokerage
   376→- real_estate: REITs, property funds, cap rates, NOI, commercial real estate
   377→- government_contracting: Federal contracts, SAM.gov, NAICS codes, procurement
   378→- investment_banking: League tables, M&A advisory, underwriting, deal teams
   379→- insurance: Carriers, reinsurers, combined ratios, AM Best ratings
   380→- pharma_biotech: Clinical trials, drug pipelines, FDA approvals, CDMOs
   381→- technology_saaS: SaaS companies, ARR/MRR, cloud infrastructure, dev tools
   382→- manufacturing: OEMs, contract manufacturers, ISO certifications, supply chain
   383→- energy_utilities: Energy companies, renewable, power generation, PPA
   384→- financial_services: FinTech, banking, payments, regulatory licenses
   385→- healthcare: Health systems, medical devices, digital health, HIPAA
   386→- education: EdTech, universities, LMS platforms, accreditation
   387→- general: Any query that doesn't fit a specialized domain
   388→
   389→CLASSIFICATION RULES:
   390→1. "Research X and write them an email" → primary: research_company, secondary: compose_outreach
   391→2. "Tell me about Stripe. Is it a good lead?" → primary: research_company, secondary: score_lead
   392→3. "Find companies like my last result" → refine_search (uses context)
   393→4. "Score it" or "Evaluate this" (after research) → score_lead (uses context of recent prospect)
   394→5. "Write to them" or "Reach out" (after research) → compose_outreach (uses context)
   395→6. "Add this to leads" → add_to_pipeline
   396→7. Simple company/person name → research_company or research_person
   397→   IMPORTANT: Single-word names after "research/find/tell me about" are almost always COMPANIES (e.g., "Research Stripe" → research_company, "Find Notion" → research_company). Multi-word names like "Patrick Collison" or "Elon Musk" are persons → research_person. When in doubt, default to research_company.
   398→8. URL starting with http → research_url
   399→9. "Compare X vs Y" or "X competitors" → analyze_competitors
   400→10. Follow-up questions about previous results → converse (but check if they want new actions)
   401→11. Domain-specific queries (mentioning "VC", "LP", "dry powder", "fund", "SEC filing", etc.) → research_company BUT set domain to the appropriate domain type. This activates the 4-phase pipeline.
   402→12. Queries requesting structured datasets, KPIs, contact matrices, or financial metrics → research_company with domain detection, even if no specific company is named.
   403→
   404→USER MESSAGE: "${userMessage}"
   405→${contextStr}
   406→

   427→/**
   428→ * Get the conversation response prompt.
   429→ * Used after actions are executed to generate a natural language response.
   430→ * Enhanced with adaptive context and proactive suggestions.
   431→ */
   432→export function getConversationResponsePrompt(
   433→  persona: AgentPersona,
   434→  intent: UserIntent,
   435→  userMessage: string,
   436→  actionResults: string,
   437→  context: ConversationContext | undefined,
   438→): string {
   439→  const contextStr = context ? `
   440→CONVERSATION CONTEXT:
   441→${context.recentProspects.length > 0 ? `- Recently discussed: ${context.recentProspects.map(p => p.companyName || p.personName).filter(Boolean).join(', ')}` : ''}
   442→${context.activeICP ? `- Active ICP: ${context.activeICP.name}` : ''}
   443→${context.lastIntent ? `- Previous intent: ${context.lastIntent}` : ''}
   444→${context.userPreferences.industries?.length ? `- User interests: ${context.userPreferences.industries.join(', ')}` : ''}
   445→` : '';
   446→
   447→  // Determine the most valuable next action based on context
   448→  let proactiveSuggestion = '';
   449→  if (intent === 'research_company' || intent === 'research_person' || intent === 'research_url') {
   450→    if (context?.activeICP) {
   451→      proactiveSuggestion = 'Since the user has an active ICP, proactively suggest scoring this prospect against it.';
   452→    } else {
   453→      proactiveSuggestion = 'Suggest composing outreach or building an ICP as the next step.';
   454→    }
   455→  } else if (intent === 'build_icp') {
   456→    proactiveSuggestion = 'Suggest testing the ICP against a real prospect or searching for matching companies.';
   457→  } else if (intent === 'score_lead') {
   458→    proactiveSuggestion = 'If the score is good, suggest composing outreach. If poor, suggest finding better matches.';
   459→  }
   460→
   461→  return `${getPersonaPrompt(persona)}
   462→
   463→CRITICAL LANGUAGE REQUIREMENT: You MUST write your ENTIRE response in English. Do NOT use Chinese or any other language. All text, descriptions, suggestions, and data references must be in English.
   464→
   465→${contextStr}
   466→
   467→USER ASKED: "${userMessage}"
   468→
   469→INTENT: ${intent}
   470→
   471→ACTION RESULTS:
   472→${actionResults}
   473→
   474→${proactiveSuggestion ? `PROACTIVE SUGGESTION: ${proactiveSuggestion}` : ''}
   475→
   476→Based on the action results above, write a conversational response to the user. Your response should:
   477→1. Be natural and conversational (not robotic) — like a knowledgeable colleague sharing findings
   478→2. Lead with the most important finding or insight — don't bury the lead
   479→3. Include specific details (names, numbers, facts) from the results — specificity builds trust
   480→4. If you found buying signals, red flags, or notable patterns, call them out explicitly
   481→5. Suggest 2-3 logical next actions the user could take, ranked by value
   482→6. Be appropriate for the ${persona} persona
   483→7. If something didn't work (missing data, failed search), explain why and suggest alternatives
   484→8. Match the depth to the user's query — detailed for specific requests, concise for simple ones
   485→
   486→Write your response directly — do not wrap it in quotes or add meta-commentary.`;
   487→}
