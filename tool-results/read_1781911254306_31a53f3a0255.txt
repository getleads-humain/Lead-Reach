   154→   154→export async function processWithOrchestrator(
   155→   155→  userMessage: string,
   156→   156→  context?: ConversationContext,
   157→   157→  forceIntent?: UserIntent,
   158→   158→  onEvent?: OrchestratorCallback,
   159→   159→  resumeFrom?: PipelineCheckpoint,
   160→   160→): Promise<{
   161→   161→  message: AgentMessage;
   162→   162→  updatedContext: ConversationContext;
   163→   163→  suggestedActions: SuggestedAction[];
   164→   164→  pipelineState: PipelineState;
   165→   165→}> {
   166→   166→  const pipelineState = createInitialPipelineState();
   167→   167→  const startTime = Date.now();
   168→   168→
   169→   169→  try {
   170→   170→    return await processWithOrchestratorInner(userMessage, context, forceIntent, pipelineState, startTime, onEvent, resumeFrom);
   171→   171→  } catch (error) {
   172→   172→    console.error('[Orchestrator] FATAL:', error);
   173→   173→
   174→   174→    const errorMsg = error instanceof Error ? error.message : 'Unknown error';
   175→   175→    pipelineState.phase = 'error';
   176→   176→
   177→   177→    // ─── GRACEFUL DEGRADE: Build a fallback response from user-supplied data ───
   178→   178→    // Even if the entire pipeline failed, we still want to:
   179→   179→    //   1. Parse the user's query for any structured data they provided
   180→   180→    //   2. Build a minimal ProspectResult from that data
   181→   181→    //   3. Render a meaningful response instead of just "Pipeline Error"
   182→   182→    //
   183→   183→    // This is the fix for the "Both stream and chat API failed" failure mode
   184→   184→    // where users saw all "Discovering..." placeholders with no data.
   185→   185→    let fallbackProspect: ProspectResult | undefined;
   186→   186→    let fallbackContent: string;
   187→   187→
   188→   188→    try {
   189→   189→      const { parseQuery } = await import('./query-parser');
   190→   190→      const parsed = parseQuery(userMessage);
   191→   191→
   192→   192→      if (parsed.signalsProvided > 0 && parsed.prepopulatedProspect) {
   193→   193→        // Build a real ProspectResult from pre-populated data
   194→   194→        fallbackProspect = {
   195→   195→          ...({
   196→   196→            queryType: parsed.guessedIntent === 'research_person' ? 'person' : 'company',
   197→   197→            query: userMessage,
   198→   198→            companyName: null, legalName: null, website: null, industry: null, subIndustry: null, description: null,
   199→   199→            hqAddress: null, city: null, stateProvince: null, country: null, postalCode: null,
   200→   200→            phoneMain: null, generalEmail: null, supportEmail: null,
   201→   201→            ceoName: null, ceoEmail: null, keyContactName: null, keyContactTitle: null, keyContactEmail: null,
   202→   202→            employeeCount: null, revenueEstimate: null, foundingYear: null, ownershipType: null,
   203→   203→            linkedinUrl: null, twitterHandle: null, facebookPage: null, techStack: [],
   204→   204→            boardMembers: [], recentNews: [], productsServices: [], partners: [], fundingInfo: null,
   205→   205→            personName: null, personTitle: null, personCompany: null, personEmail: null,
   206→   206→            personPhone: null, personLinkedin: null, personBio: null,
   207→   207→            sources: [], dataCompleteness: 0,
   208→   208→          } as ProspectResult),
   209→   209→          ...(parsed.prepopulatedProspect as Record<string, unknown>),
   210→   210→        } as unknown as ProspectResult;
   211→   211→
   212→   212→        // Calculate completeness based on what we have
   213→   213→        fallbackProspect.dataCompleteness = Math.max(
   214→   214→          fallbackProspect.dataCompleteness || 0,
   215→   215→          Math.min(70, parsed.signalsProvided * 8), // cap at 70% since research is incomplete
   216→   216→        );
   217→   217→
   218→   218→        // Build a user-friendly fallback message
   219→   219→        const filledFields: string[] = [];
   220→   220→        if (fallbackProspect.personName) filledFields.push(`**Name:** ${fallbackProspect.personName}`);
   221→   221→        if (fallbackProspect.personTitle) filledFields.push(`**Title:** ${fallbackProspect.personTitle}`);
   222→   222→        if (fallbackProspect.personCompany || fallbackProspect.companyName) {
   223→   223→          filledFields.push(`**Company:** ${fallbackProspect.personCompany || fallbackProspect.companyName}`);
   224→   224→        }
   225→   225→        if (fallbackProspect.personEmail || fallbackProspect.generalEmail) {
   226→   226→          filledFields.push(`**Email:** ${fallbackProspect.personEmail || fallbackProspect.generalEmail}`);
   227→   227→        }
   228→   228→        if (fallbackProspect.personLinkedin) filledFields.push(`**LinkedIn:** ${fallbackProspect.personLinkedin}`);
   229→   229→        if (fallbackProspect.city || fallbackProspect.country) {
   230→   230→          filledFields.push(`**Location:** ${[fallbackProspect.city, fallbackProspect.country].filter(Boolean).join(', ')}`);
   231→   231→        }
   232→   232→        if (fallbackProspect.industry) filledFields.push(`**Industry:** ${fallbackProspect.industry}`);
   233→   233→        if (fallbackProspect.personBio) filledFields.push(`**Bio:** ${fallbackProspect.personBio.slice(0, 200)}${fallbackProspect.personBio.length > 200 ? '...' : ''}`);
   234→   234→
   235→   235→        fallbackContent = `I extracted the following information from your query. However, my external research services encountered an error (${errorMsg.slice(0, 100)}), so I couldn't enrich this with additional web data. Here's what you provided:\n\n${filledFields.join('\n')}\n\nYou can try running this query again in a moment, or use the data above as-is.`;
   236→   236→      } else {
   237→   237→        fallbackContent = `I encountered an error while processing your request. Let me try a different approach.\n\n**Error:** ${errorMsg.slice(0, 150)}\n\nYou can try:\n• Rephrasing your query (e.g., "Research Stripe")\n• Asking a more specific question`;
   238→   238→      }
   239→   239→    } catch (parseErr) {
   240→   240→      console.error('[Orchestrator] Fallback parser also failed:', parseErr);
   241→   241→      fallbackContent = `I encountered an error while processing your request. Let me try a different approach.\n\n**Error:** ${errorMsg.slice(0, 150)}\n\nYou can try:\n• Rephrasing your query (e.g., "Research Stripe")\n• Asking a more specific question`;
   242→   242→    }
   243→   243→
   244→   244→    return {
   245→   245→      message: {
   246→   246→        id: `agent-fallback-${Date.now()}`,
   247→   247→        role: 'assistant',
   248→   248→        content: fallbackContent,
   249→   249→        timestamp: new Date(),
   250→   250→        persona: 'navigator',
   251→   251→        thinking: {
   252→   252→          persona: 'navigator',
   253→   253→          intent: 'converse',
   254→   254→          reasoning: `Fallback after pipeline error: ${errorMsg.slice(0, 100)}`,
   255→   255→          plan: ['Error recovery', 'Render user-supplied data'],
   256→   256→          confidence: 0.1,
   257→   257→        },
   258→   258→        actions: [{ type: 'converse', label: 'Error', status: 'failed', message: errorMsg.slice(0, 100) }],
   259→   259→        prospectData: fallbackProspect,
   260→   260→      },
   261→   261→      updatedContext: context || { recentProspects: [], activeICP: null, lastIntent: null, lastPersona: null, userPreferences: {} },
   262→   262→      suggestedActions: [
   263→   263→        { label: 'Try Again', prompt: userMessage, icon: 'RefreshCw' },
   264→   264→        { label: 'Help', prompt: 'What can you do?', icon: 'Lightbulb' },
   265→   265→      ],
   266→   266→      pipelineState,
   267→   267→    };
   268→   268→  }
   269→   269→}

   330→   330→      thinkTimeMs: null,
   331→   331→      ...update,
   332→   332→    };
   333→   333→  }
   334→   334→  emit(onEvent, { type: 'agent_status', data: { agent: agentKey, state: pipelineState.agents[agentKey] } });
   335→   335→}

   342→   342→function buildRichFallbackResponse(params: {
   343→   343→  intent: UserIntent;
   344→   344→  prospect?: ProspectResult;
   345→   345→  icp?: ICPResult;
   346→   346→  market?: MarketResult;
   347→   347→  score?: ScoreResult;
   348→   348→  outreach?: OutreachResult;
   349→   349→  userMessage: string;
   350→   350→}): string {
   351→   351→  const { intent, prospect, icp, market, score, outreach, userMessage } = params;
   352→   352→
   353→   353→  // Prospect-based responses
   354→   354→  if (prospect) {
   355→   355→    return buildFallbackResponse(prospect, intent);
   356→   356→  }
   357→   357→
   358→   358→  // Market analysis fallback
   359→   359→  if (market) {
   360→   360→    const parts: string[] = ['## Market Analysis'];
   361→   361→    if (market.summary) parts.push(market.summary);
   362→   362→    if (market.keyFindings?.length) {
   363→   363→      parts.push('\n**Key Findings:**');
   364→   364→      market.keyFindings.slice(0, 5).forEach(f => parts.push(`- ${f}`));
   365→   365→    }
   366→   366→    if (market.competitors?.length) {
   367→   367→      parts.push('\n**Competitors:**');
   368→   368→      market.competitors.slice(0, 5).forEach(c => parts.push(`- **${c.name}**: ${c.description?.slice(0, 80) || 'No description'}`));
   369→   369→    }
   370→   370→    if (market.trends?.length) {
   371→   371→      parts.push('\n**Trends:**');
   372→   372→      market.trends.slice(0, 3).forEach(t => parts.push(`- ${t}`));
   373→   373→    }
   374→   374→    return parts.join('\n\n');
   375→   375→  }
   376→   376→
   377→   377→  // Score fallback
   378→   378→  if (score) {
   379→   379→    const parts: string[] = ['## Lead Score'];
   380→   380→    parts.push(`**Overall Score:** ${score.overallScore}/100 — **${score.tier.toUpperCase()}** tier`);
   381→   381→    parts.push(`**Recommendation:** ${score.recommendation}`);
   382→   382→    if (score.dimensions) {
   383→   383→      const dims = Object.entries(score.dimensions) as [string, { score: number; reasoning: string }][];
   384→   384→      parts.push('\n**Dimension Breakdown:**');
   385→   385→      dims.forEach(([key, val]) => {
   386→   386→        parts.push(`- **${key}**: ${val.score}/100 — ${val.reasoning?.slice(0, 60) || ''}`);
   387→   387→      });
   388→   388→    }
   389→   389→    return parts.join('\n\n');
   390→   390→  }
   391→   391→
   392→   392→  // Outreach fallback
   393→   393→  if (outreach) {
   394→   394→    const parts: string[] = ['## Outreach Message'];
   395→   395→    if (outreach.subject) parts.push(`**Subject:** ${outreach.subject}`);
   396→   396→    parts.push(`**Channel:** ${outreach.channel}`);
   397→   397→    parts.push(`**Tone:** ${outreach.tone}`);
   398→   398→    parts.push(`\n${outreach.body}`);
   399→   399→    if (outreach.personalizationHooks?.length) {
   400→   400→      parts.push('\n**Personalization Hooks:**');
   401→   401→      outreach.personalizationHooks.forEach(h => parts.push(`- ${h}`));
   402→   402→    }
   403→   403→    return parts.join('\n\n');
   404→   404→  }
   405→   405→
   406→   406→  // ICP fallback
   407→   407→  if (icp) {
   408→   408→    const parts: string[] = ['## Ideal Customer Profile'];
   409→   409→    parts.push(`**${icp.name}**`);
   410→   410→    if (icp.description) parts.push(icp.description);
   411→   411→    if (icp.firmographic) {
   412→   412→      parts.push(`\n**Firmographics:** Industries: ${icp.firmographic.industries?.join(', ') || 'N/A'} | Size: ${icp.firmographic.companySizes?.join(', ') || 'N/A'} | Revenue: ${icp.firmographic.revenueRange || 'N/A'}`);
   413→   413→    }
   414→   414→    return parts.join('\n\n');
   415→   415→  }
   416→   416→
   417→   417→  // Generic fallback
   418→   418→  return `I've processed your request about "${userMessage.slice(0, 50)}". The research pipeline has completed. Check the results below for details, and let me know if you'd like me to take further action.`;
   419→   419→}

  1282→  1282→
  1283→  1283→  // Generate suggested actions

  1286→  1286→  // Final comm message

  1291→  1291→

  1306→  1306→function calculateQuickCompleteness(prospect: ProspectResult): number {
  1307→  1307→  const fields: Array<{ key: keyof ProspectResult; weight: number }> = [
  1308→  1308→    { key: 'companyName', weight: 10 },
  1309→  1309→    { key: 'website', weight: 8 },
  1310→  1310→    { key: 'industry', weight: 8 },
  1311→  1311→    { key: 'description', weight: 6 },
  1312→  1312→    { key: 'ceoName', weight: 7 },
  1313→  1313→    { key: 'employeeCount', weight: 6 },
  1314→  1314→    { key: 'revenueEstimate', weight: 6 },
  1315→  1315→    { key: 'generalEmail', weight: 6 },
  1316→  1316→    { key: 'phoneMain', weight: 5 },
  1317→  1317→    { key: 'linkedinUrl', weight: 5 },
  1318→  1318→    { key: 'city', weight: 4 },
  1319→  1319→    { key: 'country', weight: 4 },
  1320→  1320→    { key: 'foundingYear', weight: 3 },
  1321→  1321→    { key: 'twitterHandle', weight: 3 },
  1322→  1322→    { key: 'fundingInfo', weight: 4 },
  1323→  1323→    { key: 'techStack', weight: 5 },
  1324→  1324→    { key: 'recentNews', weight: 5 },
  1325→  1325→    { key: 'productsServices', weight: 5 },
  1326→  1326→  ];
  1327→  1327→
  1328→  1328→  let total = 0;
  1329→  1329→  let maxTotal = 0;
  1330→  1330→  for (const { key, weight } of fields) {
  1331→  1331→    maxTotal += weight;
  1332→  1332→    const val = prospect[key];
  1333→  1333→    if (val !== null && val !== undefined && val !== '' && !(Array.isArray(val) && val.length === 0)) {
  1334→  1334→      total += weight;
  1335→  1335→    }
  1336→  1336→  }
  1337→  1337→  return Math.round((total / maxTotal) * 100);
  1338→  1338→}
